def cost(market, p):
    return p.cost_major if market in MAJORS else p.cost_other


def wilder_atr(bars, n=14):
    """Wilder ATR: 첫 n개 TR 평균으로 시작해 (이전×(n−1)+TR)/n."""
    trs = [max(b['h'] - b['l'], abs(b['h'] - a['c']), abs(b['l'] - a['c'])) for a, b in zip(bars, bars[1:])]
    if len(trs) < n:
        return None
    atr = sum(trs[:n]) / n
    for tr in trs[n:]:
        atr = (atr * (n - 1) + tr) / n
    return atr


def relative_strength(bars, L):
    if len(bars) <= L or bars[-1 - L]['c'] <= 0:
        return None
    return bars[-1]['c'] / bars[-1 - L]['c'] - 1


def market_index(bars_by_market, n, min_markets=20):
    """종목들의 최근 n개 봉 상승률의 단순 평균(균등지수 수익률). 자료가 부족·불일치하면 None.

    비정상 값(NaN·무한대·0 이하)은 제외한다. 대다수 종목과 마지막 봉 시각 또는 n봉 전 봉 시각이 다른 종목(결측·정지)도 제외한다.
    """
    if type(n) is not int or n < 1 or type(min_markets) is not int or min_markets < 1:
        raise ValueError('regime_bars와 regime_min_markets는 1 이상의 정수여야 합니다')
    rows = []
    for bars in bars_by_market.values():
        if len(bars) <= n:
            continue
        try:
            last, first = float(bars[-1]['c']), float(bars[-1 - n]['c'])
        except (KeyError, TypeError, ValueError):
            continue
        if not (math.isfinite(last) and math.isfinite(first)) or last <= 0 or first <= 0:
            continue
        rows.append(((bars[-1].get('t'), bars[-1 - n].get('t')), last / first - 1))
    if not rows:
        return None
    counts = {}
    for key, _ in rows:
        counts[key] = counts.get(key, 0) + 1
    common = max(counts, key=counts.get)
    values = [r for key, r in rows if key == common]
    return sum(values) / len(values) if len(values) >= min_markets else None


def rank(bars_by_market, p):
    """적격 종목을 상대강도 내림차순(동률은 코드순)으로 정렬. 자료 부족 종목은 제외."""
    rows = []
    for m, bars in bars_by_market.items():
        rs, atr = relative_strength(bars, p.L), wilder_atr(bars)
        if rs is None or atr is None:
            continue
        rows.append({'market': m, 'rs': rs, 'atr': atr, 'close': bars[-1]['c'], 'bar': bars[-1]['t']})
    rows.sort(key=lambda r: (-r['rs'], r['market']))
    for i, r in enumerate(rows, 1):
        r['rank'] = i
    return rows


def chase_block(r, p):
    """과열 추격 매수 보류 사유. 후보 r(rs 포함)가 기간 상승률 상한 이상이면 문자열, 아니면 None."""
    if p.chase_rs_cap > 0 and r['rs'] >= p.chase_rs_cap:
        return f'기간 상승 {r["rs"]:+.1%} ≥ 과열 상한 {p.chase_rs_cap:.0%}'
    return None


def cap_stop(entry, stop, p):
    """손절가가 기준가에서 stop_max_pct보다 멀면 그 거리로 당긴다(올리기만 함)."""
    if p.stop_max_pct > 0:
        return max(stop, entry * (1 - p.stop_max_pct))
    return stop


def reentry_block(r, last, now, p):
    """직전 매도 기록 last={'time','price','profit'}가 있으면 재진입을 더 까다롭게 본다. 막으면 사유 문자열, 아니면 None.
    수익 청산: reentry_profit_hours 뒤부터, 그리고 reentry_window_hours 안에서는 현재가가 매도가보다 reentry_dip_atr×ATR 이상 낮을 때만(고점에서 팔고 눌리면 다시 잡기).
    손실 청산: reentry_loss_hours 뒤부터."""
    if not last:
        return None
    if rotation.enabled(p):
        why = rotation.fresh_reentry(last, p, rotation.row_close(r, p.tf))
        if why: return why
    hours = (now - last['time']) / 3600
    if hours >= p.reentry_window_hours:
        return None
    if last.get('profit'):
        if hours < p.reentry_profit_hours:
            return f'수익 청산 후 재진입 대기 {hours:.1f}/{p.reentry_profit_hours:g}시간'
        atr_pct = r.get('atr', 0) / r['close'] if r.get('close') else 0
        limit = last['price'] * (1 - p.reentry_dip_atr * atr_pct)
        if p.reentry_dip_atr > 0 and r.get('close', 0) > limit:
            return f'수익 청산 후 재진입은 매도가 {last["price"]:,.4g}보다 {p.reentry_dip_atr:g}ATR 이상 눌려야 함(현재 {r["close"]:,.4g} > {limit:,.4g})'
        return None
    if hours < p.reentry_loss_hours:
        return f'손실 청산 후 재진입 대기 {hours:.1f}/{p.reentry_loss_hours:g}시간'
    return None


def plan(ranked, held, banned, entry_ok, p):
    """15번 교체 규칙. held: 보유 종목, banned: 재진입 금지, entry_ok: 신규 진입 가능한 종목 집합.

    반환: (매도 [(종목, 사유)], 매수 후보 [종목], 판단 보류 [종목])
    """
    pos = {r['market']: r['rank'] for r in ranked}
    sells, unknown = [], []
    for m in held:
        if m not in pos:
            unknown.append(m)          # 자료 부족: 순위로 판단하지 않고 유지 (누락 처리 17번 미합의)
        elif pos[m] > p.M:
            sells.append((m, '순환'))
    keep = [m for m in held if m not in {s[0] for s in sells}]
    free = p.slots - len(keep)
    buys = []
    scan = min(p.entry_scan or p.entry_top, p.M) if rotation.enabled(p) else p.entry_top
    want = p.entry_want or p.entry_top
    for r in ranked[:scan]:
        if rotation.enabled(p) and (type(r.get('rank')) is not int or not 1 <= r['rank'] <= p.M):
            continue
        if len(buys) >= min(free, want):
            break
        m = r['market']
        if m in held or m in banned or m not in entry_ok:
            continue
        if p.require_positive_rs and r['rs'] <= 0:
            continue
        if chase_block(r, p):
            continue
        buys.append(m)
    return sells, buys, unknown


def size(equity, invested, cash, P, S, p):
    """16·27번 수량: min(계획위험/(P−S), 종목 상한, 잔여 총투입, 수수료 포함 가용현금). 최소주문 미달은 0."""
    if S <= 0 or P <= S:
        return 0.0
    qty = min(p.risk * equity / (P - S),
              p.coin_cap * equity / P,
              max(0.0, p.total_cap * equity - invested) / P,
              cash / (P * (1 + p.fee)))
    qty = math.floor(max(qty, 0.0) * 1e8) / 1e8
    return qty if qty * P >= p.min_order else 0.0


def ratchet(market, pos, price, p):
    """EXIT-T1. 고점·손절가를 갱신하고 청산 사유('수익 반납'|'추적 손절'|'손절')를 반환한다. 손절가는 내려가지 않는다.

    pos: entry, stop, atr 필요. 갱신값 peak, stop0(초기 손절가), stop을 pos에 기록한다.
    trail_enabled=False이면 기존 규칙(고정 손절)과 동일하다.
    """
    pos.setdefault('stop0', pos['stop'])
    if not p.trail_enabled:
        return '손절' if price <= pos['stop'] else None
    E, A = pos['entry'], pos['atr']
    R = p.k * A
    peak = max(pos.get('peak', E), price)
    pos['peak'] = peak
    if R > 0:
        g = (peak - E) / R
        stop = pos['stop']
        if g >= p.trail_lock_r:
            stop = max(stop, E * (1 + 2 * p.fee + cost(market, p)))
        if g >= p.trail_start_r:
            stop = max(stop, peak - p.trail_m * A)
        pos['stop'] = stop
        if p.giveback > 0 and g >= p.trail_start_r and peak > E and (peak - price) / (peak - E) >= p.giveback:
            return '수익 반납'
    if price <= pos['stop']:
        return '추적 손절' if pos['stop'] > pos['stop0'] else '손절'
    return None


def bar_after_entry(bar, entry_iso):
    """1분봉이 진입 시각 이후에 마감됐는지. 체결 이전·같은 봉 정보를 쓰지 않기 위한 검사."""
    import datetime as dt
    end = dt.datetime.fromisoformat(bar['t']) + dt.timedelta(minutes=1)
    return end > dt.datetime.fromisoformat(entry_iso)

def current_slot(now):
    return int(now // SLOT_SECONDS) * SLOT_SECONDS


def slot_due(now, last_slot):
    """지금 만들어야 할 슬롯 시작 시각. 이미 처리했거나 시작 전·허용 지연 밖이면 None."""
    slot = current_slot(now)
    if slot > last_slot and SLOT_DELAY <= now - slot <= SLOT_MAX_LATE:
        return slot
    return None


def load_last_slot(now, path=None):
    """영속 기록된 마지막 슬롯. 파일이 없거나 손상·비정각·미래 값이면 현재 슬롯을 처리한 것으로 간주한다(중복 주문안보다 한 슬롯 건너뛰는 쪽이 안전).

    가동 시작에는 항상 현재 슬롯도 처리한 것으로 본다: 재시작이 허용 지연창 안에 있어도 새 판단을 만들지 않고 다음 정규 슬롯을 기다린다."""
    path = path or SLOT_FILE
    now_slot = current_slot(now)
    try:
        value = json.loads(path.read_text(encoding='utf-8'))['slot']
        if type(value) in (int, float) and math.isfinite(value) and value % SLOT_SECONDS == 0 and 0 <= value <= now_slot:
            return max(int(value), now_slot)
    except (OSError, ValueError, KeyError, TypeError, OverflowError):
        pass
    return now_slot


def plan_tick(now, last_slot, ready, claim, make):
    """한 번의 반복에서 할 일. (새 last_slot, 상태). 신호가 아직 최신이 아니면 기록하지 않고 허용 지연 안에서 다시 시도한다."""
    slot = slot_due(now, last_slot)
    if slot is None:
        return last_slot, None
    if not ready():
        return last_slot, 'waiting'
    claim(slot)  # 먼저 기록: 이후 어떤 장애가 나도 이 슬롯은 한 번만
    try:
        make()
    except Exception:
        traceback.print_exc()
        return slot, 'failed'
    return slot, 'made'

            reasons.append('장세 필터 임시 적용 기간 만료 → 신규 매수 전부 보류 (재검토·재결정 전에는 재개하지 않음)')
        elif type(market_index) not in (int, float) or not math.isfinite(market_index):
            regime_closed = True
            reasons.append('장세 필터: 시장 지수 자료 부족·비정상 → 신규 매수 보류')
        elif market_index <= 0:
            regime_closed = True
            reasons.append(f'장세 필터: 시장 균등지수 {base.regime_bars * base.tf // 60}시간 {market_index:+.2%} ≤ 0 → 신규 매수 보류 (보유 관리·매도는 그대로)')
        else:
            reasons.append(f'장세 필터 통과: 시장 균등지수 {market_index:+.2%} > 0')
    for r in ([] if regime_closed else ranked[:scan]):
        if rotation.enabled(base) and (type(r.get('rank')) is not int or not 1 <= r['rank'] <= base.M):
            continue
        if len(picks) >= want:
            break
        if r['market'] in originally_held or r['market'] in sell_queued:
            if r['market'] in sell_queued:
                reasons.append(f'{r["market"][4:]} 동일 주문안 매도 예정: 재매수 금지')
            continue
        if r['market'] in entry_ok and r['rs'] > 0:
            last = (last_sells or {}).get(r['market'])
            why = st.chase_block(r, base) or (st.reentry_block(r, last, time.time(), base) if last else None)
            if why:
                reasons.append(f'{r["market"][4:]} 매수 보류: {why}')
            else:
                picks.append(r)
    buys = []
    # 전략 설계의 비율 한도: 종목당 coin_cap, 총투입 total_cap (계좌 평가액 기준). 고정 금액 한도를 풀어도 이 한도는 유지한다(D-29).
    equity = sum(holdings.values()) + krw
    invested = sum(v for m, v in holdings.items() if m not in sells)
    total_room = max(0.0, base.total_cap * equity - invested)
    for i, r in enumerate(picks):
        coin_room = max(0.0, base.coin_cap * equity - holdings.get(r['market'], 0.0))
        risk_room = (base.entry_risk or base.risk) * equity / base.stop_max_pct if base.size_by_risk and base.stop_max_pct > 0 else float('inf')
        amt = int(min(caps['max_order_krw'], room, cash / (len(picks) - i), coin_room, total_room, risk_room) // 1000 * 1000)
        if amt < 5000:
            reasons.append(f'{r["market"][4:]} 매수 불가: 금액 부족 (누적 한도 잔여 {room:,.0f}원, 종목 한도 잔여 {coin_room:,.0f}원, 총투입 한도 잔여 {total_room:,.0f}원)')
            continue
        buys.append((r['market'], amt))
        cash -= amt
        room -= amt
        total_room -= amt
        reasons.append(f'{r["market"][4:]} 매수 {amt:,}원: 순위 {r["rank"]}위, 상대강도 {r["rs"]:+.1%}')
    parts = [f'매도 {m[4:]} 100%' for m in sells] or ['매도 없음']
    parts += [f'매수 {m[4:]} {a}원' for m, a in buys] or ['매수 없음']
    return '실행 주문안: ' + '; '.join(parts), reasons


bar_after_entry = st.bar_after_entry


def variants(base):
    out = {}
    for L, k, M in itertools.product(GRID['L'], GRID['k'], GRID['M']):
        out[f'L{L}_k{k:g}_M{M}'] = replace(base, L=L, k=k, M=M)
    # EXIT-T1 비교군 (토론 안건 26 합의). 비교 기준 '기존'은 위 격자의 기본 변형(trail 꺼짐)이다.
    inf = float('inf')
    out['EXIT_T1'] = replace(base, trail_enabled=True, trail_start_r=inf, giveback=0.0)
    out['EXIT_T1T2'] = replace(base, trail_enabled=True, giveback=0.0)
    out['EXIT_T1T2T3'] = replace(base, trail_enabled=True)
    return out

def metrics(folder, A0):
    def rows(name):
        f = folder / name
        return [json.loads(x) for x in f.read_text(encoding='utf-8').splitlines() if x.strip()] if f.exists() else []
    trades, eq = rows('trades.jsonl'), rows('equity.jsonl')
    closed = [t for t in trades if t['side'] == '매도']
    wins = [t['pnl'] for t in closed if t['pnl'] > 0]
    losses = [-t['pnl'] for t in closed if t['pnl'] <= 0]
    peak, mdd = A0, 0.0
    for r in eq:
        peak = max(peak, r['P'])
        mdd = max(mdd, 1 - r['P'] / peak)
    last = eq[-1] if eq else {'P': A0, 'H': None}
    h_ret = None
    if eq and eq[0].get('H') and last.get('H'):
        h_ret = last['H'] / eq[0]['H'] - 1
    return {'ret': last['P'] / A0 - 1, 'h_ret': h_ret, 'buys': sum(t['side'] == '매수' for t in trades),
            'closed': len(closed), 'win': len(wins) / len(closed) if closed else None,
            'pf': (sum(wins) / sum(losses)) if losses else (None if not wins else float('inf')),
            'avg': sum(t['pnl_pct'] for t in closed) / len(closed) if closed else None,
            'mdd': mdd, 'stops': sum(t.get('reason') == '손절' for t in closed), 'since': eq[0]['time'] if eq else ''}


def fmt(v, kind):
    if v is None:
        return '-'
    if kind == '%':
        return f'{v * 100:+.2f}%'
    if kind == 'pf':
        return '∞' if v == float('inf') else f'{v:.2f}'
    return str(v)

               '기존 보유도 신규 후보와 같은 기준으로 평가했다(사용자 지시). 두 AI가 아래 문구를 그대로 수용해 합의하면 봇이 실행한다. '
               '수정하려면 근거를 대고 고친 문구로 합의한다.\n\n'
               f'`{prop["text"]}`\n\n' + '\n'.join(f'- {x}' for x in prop['reasons']) + '\n')

def _md_to_html(md):
    for line in md.splitlines():
        if line.startswith('### '):
            yield f'<h2>{html.escape(line[4:])}</h2>'
        elif line.startswith('|---'):
            continue
        elif line.startswith('|'):
            cells = [c.strip() for c in line.strip('|').split('|')]
            tag = 'th' if cells[0] == '모델' else 'td'
            yield '<tr>' + ''.join(f'<{tag}>{html.escape(c)}</{tag}>' for c in cells) + '</tr>'
        elif line.strip():
            yield f'<p>{html.escape(line)}</p>'
    yield ''


def share(md):
    """근거자료의 성과 구역만 최신 내용으로 바꾼다(파일이 계속 커지지 않게)."""
    try:
        text = EVIDENCE.read_text(encoding='utf-8')
        block = f'{MARK_A}\n{md}{MARK_B}'
        if MARK_A in text and MARK_B in text:
            text = re.sub(re.escape(MARK_A) + r'.*?' + re.escape(MARK_B), lambda _: block, text, flags=re.S)
        else:
            text += '\n\n' + block + '\n'
        EVIDENCE.write_text(text, encoding='utf-8')
    except OSError:
        pass

class Shadow:
    def __init__(self):
        self.base, self.cfg = load_params()
        self.vars = variants(self.base)
        self.ledgers = {n: Ledger(SHADOW / n) for n in self.vars}
        self.markets = {}
        self.prop = None
        self.base_ranked, self.rs_out, self.entry_ok = [], {}, set()
        self.signal_closed_at = self.signal_tf_minutes = self.evaluated_at = None
        self.rotation_counts = {}

    def proposal_signal(self, now=None):
        """Return provenance only for the current completed signal, never a relabelled cache."""
        now = time.time() if now is None else now
        tf = self.base.tf
        if type(tf) is not int or tf <= 0 or self.signal_tf_minutes != tf:
            raise ValueError('주문안 신호 주기 불일치: 새 판단 필요')
        closed, evaluated = self.signal_closed_at, self.evaluated_at
        if any(type(value) not in (int, float) or not math.isfinite(value)
               for value in (now, closed, evaluated)):
            raise ValueError('주문안 신호 출처 누락 또는 비정상 시각')
        expected = boundary(now - SETTLE, tf * 60)
        if closed != expected or not closed + SETTLE <= evaluated <= now:
            raise ValueError('주문안 신호가 최신 마감 판단과 다름: 오래된 순위 사용 금지')
        if not self.base_ranked:
            raise ValueError('주문안 순위 자료 없음')
        for row in self.base_ranked:
            try:
                bar_close = dt.datetime.fromisoformat(row['bar']).timestamp() + tf * 60
            except (KeyError, TypeError, ValueError, OverflowError) as error:
                raise ValueError('주문안 순위 봉 시각 오류') from error
            if bar_close != closed:
                raise ValueError('주문안 순위와 신호 마감 시각 불일치')
        return {'signal_closed_at': closed, 'signal_tf_minutes': tf, 'evaluated_at': evaluated}

    def make_proposal(self):
        self.proposal_signal()  # Refuse stale cached ranks before any private account lookup.
        if rotation.enabled(self.base):
            if self.cfg.get('live', {}).get('slot_plans') is not True:
                raise ValueError('새 순환 정책은 고정 슬롯 경로만 허용')
            rotation.envelope(self.base, self.signal_closed_at, time.time())
        rows = upbit.accounts()
        krw = sum(float(r['balance']) for r in rows if r['currency'] == 'KRW')
        coins = {'KRW-' + r['currency']: float(r['balance']) for r in rows
                 if r['currency'] != 'KRW' and 'KRW-' + r['currency'] in self.markets}
        tick = upbit.tickers(coins) if coins else {}
        holdings = {m: q * tick[m]['trade_price'] for m, q in coins.items() if m in tick}
        caps = {'max_order_krw': 30000, 'max_total_buy_krw': 100000, 'max_sell_krw_per_plan': 300000, **self.cfg.get('live', {})}
        lf = DATA / 'live' / 'live.json'
        state = json.loads(lf.read_text(encoding='utf-8')) if lf.exists() else {}
        bought = state.get('bought_krw', 0)
        caps.update(state.get('limits', {}))
        text, reasons = proposal(holdings, self.base_ranked, self.rs_out, self.entry_ok, self.base, caps, krw, bought, state.get('last_sell'),
                                 market_index=getattr(self, 'market_index', None),
                                 rotation_state=getattr(self, 'rotation_counts', None),
                                 rotation_slot=self.signal_closed_at)
        # Account/quote lookups may cross a candle boundary. Recheck the actual source before writing.
        provenance = self.proposal_signal()
        # Public signal prices, not account inventory: the receiver can enforce the SAME
        # chase cap against a later executable ask instead of trusting a frozen return.
        entry_context = {}
        for row in self.base_ranked:
            close, rs = row.get('close'), row.get('rs')
            if (type(close) not in (int, float) or not math.isfinite(close) or close <= 0
                    or type(rs) not in (int, float) or not math.isfinite(rs) or rs <= -1):
                continue
            anchor = close / (1 + rs)
            if not math.isfinite(anchor) or anchor <= 0: continue
            entry_context[row['market']] = {'market': row['market'], 'signal_close': close,
                'anchor_close': anchor, 'rs': rs, **provenance}
            if rotation.enabled(self.base):
                entry_context[row['market']].update(rank=row['rank'], policy_hash=rotation.policy_hash(self.base))
        self.prop = {'time': now_iso(), 'text': text, 'reasons': reasons,
                     'entry_context': entry_context, 'market_index': getattr(self, 'market_index', None), **provenance}
        if rotation.enabled(self.base):
            self.prop['rotation'] = rotation.envelope(self.base, self.signal_closed_at, time.time())
            self.prop['rotation_markets'] = {r['market']: {k:r[k] for k in ('rank', 'rs', 'close', 'atr', 'bar')} for r in self.base_ranked}
            rotation.append_audit(SHADOW / 'rotation_audit.jsonl',
                {'event': 'decision_created', 'rotation': self.prop['rotation'],
                 'markets': self.prop['rotation_markets'], 'reasons': reasons, 'plan': text})
        tmp = DATA / 'proposal.tmp'
        tmp.write_text(json.dumps(self.prop, ensure_ascii=False, indent=1), encoding='utf-8')
        tmp.replace(DATA / 'proposal.json')

    def init(self):
        src = json.loads((DATA / 'ALL' / 'ledger.json').read_text(encoding='utf-8'))
        for n, lg in self.ledgers.items():
            if not lg.ready:
                H = src['H']
                lg.start(src['E0'], 1.0, H['basket'], H['krw'], H['unpriced'], label=f'그림자 {n}')
                lg.d['last_4h'] = boundary(time.time() - SETTLE, self.base.tf * 60)
                lg.save()

    def decide(self, end):
        # A failed refresh must not leave the previous ranks eligible for a newly stamped order.
        self.signal_closed_at = self.signal_tf_minutes = self.evaluated_at = None
        self.prop = None
        self.markets = upbit.krw_markets()
        tick = upbit.tickers(self.markets)
        pool = sorted((m for m in tick if m not in self.base.exclude), key=lambda m: -tick[m]['acc_trade_price_24h'])
        universe = pool[:self.base.universe]
        held = {m for lg in self.ledgers.values() for m in lg.d['positions']}
        try:
            held |= {'KRW-' + r['currency'] for r in upbit.accounts() if 'KRW-' + r['currency'] in self.markets}
        except Exception:
            pass
        bars = {}
        for m in dict.fromkeys(universe + sorted(held)):
            b = upbit.completed_candles(m, self.base.tf, 200)
            if b and dt.datetime.fromisoformat(b[-1]['t']).timestamp() + self.base.tf * 60 == end:
                bars[m] = b
            time.sleep(0.12)
        uni = set(universe)
        entry_ok = {m for m in universe if m in bars and not self.markets.get(m, {}).get('warning')}
        self.base_ranked = st.rank({m: b for m, b in bars.items() if m in uni}, self.base)
        self.market_index = st.market_index({m: b for m, b in bars.items() if m in uni}, self.base.regime_bars,
                                            self.base.regime_min_markets)
        self.rs_out = {m: st.relative_strength(b, self.base.L) for m, b in bars.items() if m not in uni}
        self.entry_ok = entry_ok
        need_books = set()
        plans = {}
        for n, lg in self.ledgers.items():
            if (lg.d['last_4h'] or 0) >= end:
                continue
            p = self.vars[n]
            ranked = st.rank({m: b for m, b in bars.items() if m in uni}, p)
            lg.d['banned'] = {m: t for m, t in lg.d['banned'].items() if t > time.time()}
            sells, buys, _ = st.plan(ranked, list(lg.d['positions']), set(lg.d['banned']), entry_ok, p)
            plans[n] = (p, ranked, sells, buys)
            need_books |= {m for m, _ in sells} | set(buys)
        books = upbit.orderbooks(sorted(need_books)) if need_books else {}
        prices = {m: t['trade_price'] for m, t in tick.items()}
        for n, (p, ranked, sells, buys) in plans.items():
            lg = self.ledgers[n]
            for m, reason in sells:
                lg.sell(m, books[m]['bid'], reason, p)
            atr = {r['market']: r['atr'] for r in ranked}
            for m in buys:
                eq, inv = lg.equity(prices), lg.invested(prices)
                if not lg.buy(m, books[m]['ask'], atr[m], lambda P, S: st.size(eq, inv, lg.d['cash'], P, S, p), p):
                    lg.d['skipped'] = lg.d.get('skipped', 0) + 1
            lg.d['signals'] = lg.d.get('signals', 0) + len(buys)
            lg.d['last_4h'] = end
            lg.save()
        self.signal_closed_at = end
        self.signal_tf_minutes = self.base.tf
        self.evaluated_at = time.time()

    def stops(self):
        held = sorted({m for lg in self.ledgers.values() for m in lg.d['positions']})
        if not held:
            return
        live = {m: t['trade_price'] for m, t in upbit.tickers(held).items()}
        # EXIT-T1: 확정 1분봉 종가로만 평가하고 봉마다 한 번만 반영한다. 진입 이후 마감된 봉만 쓴다.
        closes = {}
        for n, lg in self.ledgers.items():
            if self.vars[n].trail_enabled:
                for m in lg.d['positions']:
                    if m not in closes:
                        try:
                            bars = upbit.completed_candles(m, 1, 3)
                        except Exception:
                            bars = []
                        closes[m] = bars[-1] if bars else None
        reasons = {}
        for n, lg in self.ledgers.items():
            p = self.vars[n]
            for m, pos in lg.d['positions'].items():
                if m not in live:
                    continue
                why = []
                if p.trail_enabled:
                    bar = closes.get(m)
                    if bar and pos.get('last_bar') != bar['t'] and bar_after_entry(bar, pos['time']):
                        pos['last_bar'] = bar['t']
                        r = st.ratchet(m, pos, bar['c'], p)
                        if r:
                            why.append(r)
                    if live[m] <= pos.get('stop0', pos['stop']) and '손절' not in why:
                        why.append('손절')  # 초기 고정 손절은 기존처럼 실시간 가격으로 보호
                elif live[m] <= pos['stop']:
                    why.append('손절')
                if why:
                    reasons[(n, m)] = '+'.join(why)
            if p.trail_enabled:
                lg.save()
        hit = sorted({m for _, m in reasons})
        if not hit:
            return
        books = upbit.orderbooks(hit)
        for (n, m), why in reasons.items():
            lg = self.ledgers[n]
            if m in lg.d['positions']:  # 종목·비교군당 청산 요청은 한 번, 사유는 모두 기록
                lg.sell(m, books[m]['bid'], why, self.vars[n], banned_until=time.time() + self.base.reentry_hours * 3600)

    def snapshot(self):
        ms = sorted({m for lg in self.ledgers.values() for m in list(lg.d['positions']) + list(lg.d['H']['basket'])})
        prices = {m: t['trade_price'] for m, t in upbit.tickers(ms).items()} if ms else {}
        for lg in self.ledgers.values():
            lg.snapshot(prices)

    def run(self):
        if rotation.enabled(self.base) and self.cfg.get('live', {}).get('slot_plans') is not True:
            raise ValueError('순환 정책/고정 슬롯 설정 불일치: 기존 경로로 전환하지 않음')
        self.init()
        if self.cfg.get('live', {}).get('enabled') and self.cfg.get('live', {}).get('auto_proposal'):
            # Recompute current ranks on restart; ledgers already processed at this close are skipped.
            self.decide(boundary(time.time() - SETTLE, self.base.tf * 60))

def parse(text):
    """'매도 ONG 100%; 매수 BTC 20000원' -> [('sell','KRW-ONG',100.0), ('buy','KRW-BTC',20000)]. 형식 오류는 ValueError."""
    orders = []
    for part in [x.strip().rstrip('.') for x in text.split(';')]:
        if not part or part in ('매도 없음', '매수 없음', '없음'):
            continue
        if '<' in part or '>' in part:
            raise ValueError(f'조건부 항목은 실행하지 않음: {part}')
        if m := SELL.match(part):
            pct = float(m[2])
            if not 0 < pct <= 100:
                raise ValueError(f'매도 비율 오류: {part}')
            orders.append(('sell', 'KRW-' + m[1], pct))
        elif m := BUY.match(part):
            orders.append(('buy', 'KRW-' + m[1], int(m[2].replace(',', ''))))
        else:
            raise ValueError(f'형식 오류: {part}')
    return orders


def latest_plan(doc):
    """합의 상태인 칸 중 번호가 가장 큰 실행 주문안 (번호, 원문)."""
    best = None
    for m in ROW.finditer(doc):
        if m[2].strip() != '합의':
            continue
        p = PLAN.search(m[3])
        if p and (best is None or int(m[1]) > best[0]):
            best = (int(m[1]), p[1].strip())
    return best


def parse_limits(text):
    """'1건 60000원; 누적 300000원' -> {'max_order_krw': 60000, 'max_total_buy_krw': 300000}"""
    out = {}
    for part in [x.strip().rstrip('.') for x in text.split(';') if x.strip()]:
        m = LIMIT_ITEM.match(part)
        if not m:
            raise ValueError(f'한도 형식 오류: {part}')
        out[LIMIT_KEYS[m[1]]] = int(m[2].replace(',', ''))
    return out

def throttle(current, wanted, ceiling):
    """올리는 값은 현재의 2배와 계좌 총평가액 중 작은 값까지. 내리는 값은 그대로."""
    out = {}
    for k, v in wanted.items():
        cur = current[k]
        out[k] = v if v <= cur else int(min(v, cur * 2, ceiling))
    return out

    # ---------- 손실 ----------
    def unrealized(self, prices):
        return sum(p['qty'] * prices[m] * (1 - 0.0005) - p['cost'] for m, p in self.d['positions'].items() if m in prices)

    def missing_valuations(self, prices):
        return [m for m in self.d['positions']
                if m not in prices or type(prices[m]) not in (int, float)
                or not math.isfinite(prices[m]) or prices[m] <= 0]

    def equity_total(self, prices=None):
        """계좌 총자산(원화 잔고+잠금+코인 평가). 조회 실패·가격 누락이 있으면 None."""
        try:
            rows = upbit.accounts()
            valid = set(upbit.krw_markets())  # 원화 시장이 없는(상장폐지 등) 자산은 시세 조회에서 제외
            coins = {'KRW-' + r['currency']: float(r['balance']) + float(r['locked']) for r in rows
                     if r['currency'] != 'KRW' and float(r['balance']) + float(r['locked']) > 0
                     and 'KRW-' + r['currency'] in valid}
            krw = sum(float(r['balance']) + float(r['locked']) for r in rows if r['currency'] == 'KRW')
            tick = upbit.tickers(list(coins)) if coins else {}
        except Exception:
            return None
        total = krw
        for m, q in coins.items():
            px = (prices or {}).get(m) or (tick.get(m) or {}).get('trade_price')
            if px: total += q * px  # 시세 없는 먼지 자산은 총자산에서 제외
        return total

    def loss_limit(self, prices=None):
        """신규 매수 중단선(원). loss_stop_pct>0이면 기준 총자산 × 비율(기준은 처음 한 번 저장), 아니면 loss_stop_krw."""
        pct = float(self.cfg.get('loss_stop_pct') or 0)
        if pct <= 0: return float(self.cfg['loss_stop_krw'])
        base = self.d.get('risk_base_equity')
        if not base:
            base = self.equity_total(prices)
            if not base: return float(self.cfg['loss_stop_krw'])  # 총자산을 못 구하면 금액 한도로 대체
            self.d['risk_base_equity'] = base
            self.save()
            self.record({'event': '손실 중단선 기준 총자산 확정', 'base': base, 'pct': pct, 'limit': base * pct})
        return base * pct

    def raw_loss(self, prices):
        return -(self.d['realized'] + self.unrealized(prices) - self.d.get('risk_baseline_unrealized', 0.))

    def resume_buying(self, prices, reason='사용자 지시'):
        """중단 해제. 지금까지의 손실을 기준(offset)으로 옮겨 손실 집계를 지금부터 다시 쌓는다(과거 손실은 장부에 남는다)."""
        if self.missing_valuations(prices):
            return False, '보유분 가격이 없어 손실 기준을 정할 수 없음'
        was = self.d['halted']
        offset = self.raw_loss(prices)
        self.d['risk_loss_offset'] = offset
        self.d['risk_day'] = dt.datetime.now(dt.timezone(dt.timedelta(hours=9))).strftime('%Y-%m-%d')
        self.d['halted'] = ''
        self.save()
        self.record({'event': '신규 매수 재개', 'reason': reason, 'loss_offset': offset, 'was_halted': was})
        self.share(f'신규 매수 재개({reason}). 손실 집계 기준을 {offset:,.0f}원으로 옮김. 직전 중단: {was or "없음"}')
        return True, f'재개: 손실 기준 {offset:,.0f}원'

    def check_resume_request(self, prices):
        """data/live/RESUME_BUY 파일이 있으면 수동 재개 요청으로 처리한다(tools/resume_trading.py가 만든다)."""
        f = self.folder / 'RESUME_BUY'
        if not f.exists():
            return None
        try:
            reason = (f.read_text(encoding='utf-8').strip() or '사용자 지시')[:100]
        except Exception:
            reason = '사용자 지시'
        ok, msg = self.resume_buying(prices, '수동 재개: ' + reason)
        if ok:
            f.unlink()
        return msg

    def check_loss(self, prices):
        missing = self.missing_valuations(prices)
        if missing:
            if self.d.get('valuation_incomplete') != missing:
                self.record({'event': '손익 평가 보류', 'markets': missing,
                             'reason': '일부 보유분 가격 누락·무효: 손익을 0으로 간주하지 않음'})
            self.d['valuation_incomplete'] = missing
            return None
        self.d.pop('valuation_incomplete', None)
        if self.cfg.get('loss_daily'):
            today = dt.datetime.now(dt.timezone(dt.timedelta(hours=9))).strftime('%Y-%m-%d')
            if self.d.get('risk_day') != today:
                was = self.d['halted']
                self.d['risk_day'] = today
                self.d['risk_loss_offset'] = self.raw_loss(prices)
                self.d['halted'] = ''
                self.save()
                self.record({'event': '손실 집계 하루 단위 시작', 'day': today, 'loss_offset': self.d['risk_loss_offset'], 'was_halted': was})
                if was:
                    self.share(f'새 날 시작: 손실 집계를 다시 시작하고 신규 매수를 재개함. 직전 중단: {was}')
        loss = self.raw_loss(prices) - self.d.get('risk_loss_offset', 0.)
        limit = self.loss_limit(prices)
        if loss >= limit and not self.d['halted']:
            pct = float(self.cfg.get('loss_stop_pct') or 0)
            self.d['halted'] = f'{now_iso()} 실거래 누적 손실 {loss:,.0f}원 ≥ 중단선 {limit:,.0f}원' + (f'(기준 총자산의 {pct:.2%})' if pct > 0 else '')
            self.save()
            self.record({'event': '신규 매수 중단', 'reason': self.d['halted']})
            self.share(f'손실 중단선 도달로 신규 매수를 중단함. {self.d["halted"]}. 기존 실거래 보유의 손절 감시는 계속. 사용자 검토 필요.')
        return loss

    # ---------- 주문 ----------
    def _wait(self, order):
        for _ in range(20):
            o = upbit.get_order(order['uuid'])
            if o['state'] in ('done', 'cancel'):
                return o
            time.sleep(0.5)
        return upbit.get_order(order['uuid'])

    @staticmethod
    def _fill(o):
        trades = o.get('trades') or []
        vol = sum(float(t['volume']) for t in trades)
        funds = sum(float(t['funds']) for t in trades)
        return vol, funds, float(o.get('paid_fee') or 0)

    def _pending(self, risk_market=None):
        return (self.d.get('risk_pending_orders', {}).get(risk_market) if risk_market is not None
                else self.d.get('pending_order'))

    def _set_pending(self, pending, risk_market=None):
        if risk_market is None:
            self.d['pending_order'] = pending
        elif pending is None:
            self.d.setdefault('risk_pending_orders', {}).pop(risk_market, None)
        else:
            self.d.setdefault('risk_pending_orders', {})[risk_market] = pending

    def has_uncertain_orders(self):
        return bool(self.d.get('pending_order') or self.d.get('risk_pending_orders'))

    def market_has_uncertain_order(self, market):
        primary = self.d.get('pending_order')
        return bool((primary and primary['market'] == market)
                    or market in self.d.get('risk_pending_orders', {}))

    def _apply_pending(self, order, risk_market=None):
        """Apply cumulative fills once and retain partial/unknown orders durably."""
        pending = self._pending(risk_market)
        if not pending: raise ValueError('정산할 주문 의도 없음')
        if order.get('market', pending['market']) != pending['market']:
            raise ValueError('주문 종목 불일치: 정산 보류')
        if pending.get('uuid') and order.get('uuid', pending['uuid']) != pending['uuid']:
            raise ValueError('주문 식별자 불일치: 정산 보류')
        if order.get('identifier', pending['identifier']) != pending['identifier']:
            raise ValueError('클라이언트 주문 식별자 불일치: 정산 보류')
        if order.get('side', pending['side']) != pending['side']:
            raise ValueError('주문 방향 불일치: 정산 보류')
        volume, funds, fee = self._fill(order)
        if any(not math.isfinite(x) or x < 0 for x in (volume, funds, fee)):
            raise ValueError('체결 수량·금액 무효: 정산 보류')
        terminal = order.get('state') in ('done', 'cancel')
        executed = float(order.get('executed_volume', volume))
        if not math.isfinite(executed) or executed < 0:
            raise ValueError('주문 누적 체결량 무효: 정산 보류')
        if terminal and executed > volume + 1e-8:
            raise ValueError('상세 체결 자료 부족: 주문 대사 필요')
        realized_before = self.d.get('realized', 0.)
        basis = None
        dv = volume - pending.get('applied_volume', 0.)
        df = funds - pending.get('applied_funds', 0.)
        dfee = fee - pending.get('applied_fee', 0.)
        if min(dv, df, dfee) < -1e-8:
            raise ValueError('주문 누적 체결량 역전: 정산 보류')
        m = pending['market']
        if dv > 1e-12:
            if pending['side'] == 'bid':
                old = self.d['positions'].get(m)
                qty = (old['qty'] if old else 0.) + dv
                cost = (old['cost'] if old else 0.) + df + dfee
                stop = st.cap_stop(df / dv, df / dv - self.p.k * pending['atr'], self.p)
                new = {'qty': qty, 'cost': cost,
                    'stop': max(old['stop'], stop) if old else stop,
                    'time': old['time'] if old else pending['time']}
                if old is None:
                    # EXIT-T1 대상: 봇이 새로 산 포지션만. E·A·초기 손절가는 첫 체결에서 확정한다(안건 26).
                    px = df / dv
                    new.update(entry=px, atr=pending['atr'], stop0=stop,
                               trail=bool(self.cfg.get('trail_enabled')) and stop < px)
                else:
                    # 추가 체결·추가 매수 lot 처리는 미합의(안건 27): 소급 변경하지 않고 EXIT-T1에서 제외한다.
                    for k in ('entry', 'atr', 'stop0', 'last_bar', 'peak', 'profit_guard_v', 'profit_guard_peak', 'profit_guard_stop', 'stop0_rebased', 'stop_rebased'):
                        if k in old: new[k] = old[k]
                    new['trail'] = False
                if rotation.enabled(self.p) and old is None:
                    new['rotation_initial_risk'] = {'per_unit': max(0., df / dv - stop),
                        'entry_price': df / dv, 'stop': stop, 'basis': 'first_fill_price_minus_initial_stop_excludes_costs'}
                self.d['positions'][m] = new
                self.d['bought_krw'] += df + dfee
            else:
                pos = self.d['positions'].get(m)
                if not pos and (risk_market is not None or self.d.get('account_adoption')):
                    raise ValueError('위임 장부 없는 매도 체결: 대사 필요')
                if pos:
                    if dv > pos['qty'] + 1e-8:
                        raise ValueError('위임 장부 수량 초과 체결: 대사 필요')
                    fraction = min(1., dv / pos['qty'])
                    basis = pos['cost'] * fraction
                    self.d['realized'] += df - dfee - basis
                    pos['qty'] = max(0., pos['qty'] - dv)
                    pos['cost'] -= basis
                    if pos['qty'] <= 1e-8:
                        # 재진입 규칙(D-38)을 위해 전량 청산 기록을 남긴다
                        self.d.setdefault('last_sell', {})[m] = {'time': time.time(), 'price': df / dv, 'profit': (df - dfee - basis) > 0,
                                                                  'reason': str(pending.get('reason', ''))[:40]}
                        del self.d['positions'][m]
        elif dfee > 0:
            if pending['side'] == 'bid':
                if m in self.d['positions']: self.d['positions'][m]['cost'] += dfee
                self.d['bought_krw'] += dfee
            else:
                self.d['realized'] -= dfee
        delta_pnl = self.d.get('realized', 0.) - realized_before
        if rotation.enabled(self.p) and pending.get('rotation_evidence', {}).get('decision'):
            state = self.d['rotation_state']
            state['net_realized'] += delta_pnl
            budget = rotation.saved_loss_budget(self.cfg, self.d)
            if budget is None:
                state['paused'] = '저장된 순환 손실 기준 무효: 정규 순환 보류'
            elif state['net_realized'] <= -budget:
                state['paused'] = '순환 시험 손실이 기존 손실 한도 도달: 정규 순환 보류, 위험 청산 유지'
        pending.update(applied_volume=volume, applied_funds=funds, applied_fee=fee)
        if order.get('uuid'): pending['uuid'] = order['uuid']
        if terminal:
            self._set_pending(None, risk_market)
        self.save()
        if dv > 1e-12 or terminal:
            self.record({'side': '매수' if pending['side'] == 'bid' else '매도', 'market': m,
                'volume': max(0., dv), 'funds': max(0., df), 'fee': max(0., dfee),
                'identifier': pending['identifier'], 'state': order.get('state'), 'reason': pending['reason']})
        if rotation.enabled(self.p) and (dv > 1e-12 or dfee > 0):
            audit = pending.get('rotation_evidence', {})
            fill_times = [t.get('created_at') for t in order.get('trades', []) if isinstance(t, dict) and t.get('created_at')]
            rotation.append_audit(self.folder / 'rotation_audit.jsonl',
                {'event': 'fill', 'market': m, 'side': pending['side'],
                 'volume': dv, 'funds': df, 'fee': dfee, 'realized_delta': delta_pnl,
                 'average_fill': df / dv if dv > 0 else None,
                 'observed_basis': basis,
                 'net_return_pct': delta_pnl / basis * 100 if basis and basis > 0 else None,
                 'r_multiple': (delta_pnl / (dv * audit.get('position_at_submit', {}).get('rotation_initial_risk', {}).get('per_unit', 0)))
                    if dv > 0 and audit.get('position_at_submit', {}).get('rotation_initial_risk', {}).get('per_unit', 0) > 0 else None,
                 'exchange_fill_times_if_provided': fill_times,
                 'exchange_order_created_at': order.get('created_at'), 'analysis': audit})
        return terminal

    def recover_pending(self, risk_market=None):
        pending = self._pending(risk_market)
        if not pending: return True
        order = upbit.get_order(pending['uuid']) if pending.get('uuid') else upbit.get_order_by_identifier(pending['identifier'])
        return self._apply_pending(order, risk_market)

    def recover_all_pending(self):
        """Unrelated market risk exits may continue while an order is unresolved."""
        slots = [None] + list(self.d.get('risk_pending_orders', {}))
        for market in slots:
            if not self._pending(market): continue
            try:
                self.recover_pending(market)
            except Exception as error:
                key = market or 'primary'
                message = type(error).__name__ + ': ' + str(error)[:250]
                alerts = self.d.setdefault('recovery_alerts', {})
                if alerts.get(key) != message:
                    alerts[key] = message
                    try:
                        self.record({'event': '미확정 주문 조회 대기', 'slot': key, 'reason': message})
                    except OSError:
                        pass
        return not self.has_uncertain_orders()

    def _trade(self, side, market, amount, reason, risk=False, entry_context=None, validate_entry=False):
        self.assert_unprotected(market)
        if not self.cfg['enabled']: raise RuntimeError('실거래 비활성')
        if risk:
            if side != 'ask': raise ValueError('위험 청산은 매도만 허용')
            if self.market_has_uncertain_order(market): raise RuntimeError('동일 종목 미확정 주문: 중복 청산 차단')
        elif self.has_uncertain_orders():
            raise RuntimeError('미확정 주문 대사 중: 중복 주문 차단')
        atr = None
        if side == 'bid':
            if validate_entry:
                why = self.entry_chase_block(market, entry_context)
                if why: raise ValueError(why)
            atr = st.wilder_atr(upbit.completed_candles(market, self.cfg['stop_tf'], 100))
            if atr is None or not math.isfinite(atr) or atr <= 0:
                raise ValueError('손절 ATR 자료 부족·무효: 매수 보류')
        checked=self.scan_manual_orders(market,risk,amount)
        if checked is not None: amount=checked
        if amount<=0: raise ProtectedAssetError('현재 가용 봇 보유 수량 없음: 주문 보류')
        if side=='bid' and validate_entry:
            # Provenance and ATR lookups precede the final executable quote check.
            book = upbit.orderbooks([market]).get(market, {})
            why = self.entry_chase_block(market, entry_context, book.get('ask'))
            if why: raise ValueError(why)
        self.assert_unprotected(market)  # ATR/quote requests may have taken time.
        pending = {'identifier': 'v5-' + uuid.uuid4().hex, 'side': side, 'market': market,
                   'amount': amount, 'reason': str(reason), 'time': now_iso(), 'atr': atr}
        risk_market = market if risk else None
        if rotation.enabled(self.p):
            meta = getattr(self, '_rotation_meta', None)
            evidence = rotation.trade_evidence(self.d['positions'].get(market), reason, time.time(), meta)
            if meta:
                evidence['intent_id'] = rotation.intent_id(meta['slot'], market, side, meta['policy_hash'])
            pending['rotation_evidence'] = evidence
        self._set_pending(pending, risk_market)
        self.save()  # Persist identifier BEFORE transmission; unknown outcomes are looked up, never replayed.
        try:
            self.assert_unprotected(market)  # Do not transmit a newly protected asset.
            order = (upbit.market_buy(market, amount, identifier=pending['identifier']) if side == 'bid'
                     else upbit.market_sell(market, amount, identifier=pending['identifier']))
        except (upbit.OrderRejected, SessionLost, ProtectedAssetError) as error:
            # Rejected submission or failed lease check happened before acceptance; do not create a phantom order.
            self._set_pending(None, risk_market)
            self.save()
            self.record({'event': '주문 전송 차단·거부', 'market': market,
                         'identifier': pending['identifier'], 'reason': str(error)[:300]})
            raise
        if order.get('uuid'):
            pending['uuid'] = order['uuid']; self.save()
        order = self._wait(order)
        if not self._apply_pending(order, risk_market):
            raise RuntimeError('주문 미종결: 체결 반영 후 신규 주문 대기')
        return order

    def poll(self, markets):
        """봇 주문안(위임)과 토론 합의 주문안을 확인하고, 새것이면 한 번 실행한다."""
        if not self.cfg['enabled']:
            return
        if not self.recover_all_pending(): return
        if self.cfg['auto_proposal']:
            self.poll_proposal(markets)
        # An adopted account uses fresh strategy proposals, never replays historical discussion orders.
        if self.d.get('account_adoption') or rotation.enabled(self.p): return
        if not DEBATE.exists():
            return
        mtime = DEBATE.stat().st_mtime
        if mtime == self.doc_mtime:
            return
        self.doc_mtime = mtime
        doc = DEBATE.read_text(encoding='utf-8')
        try:
            self.apply_limits(doc)
        except Exception as ex:
            self.record({'event': '한도 변경 실패', 'reason': f'{type(ex).__name__}: {ex}'})
        plan = latest_plan(doc)
        if not plan:
            return
        key = hashlib.sha256(f'{plan[0]}|{plan[1]}'.encode()).hexdigest()[:16]
        if key in self.d['executed']:
            return
        self.d['executed'][key] = {'id': plan[0], 'text': plan[1], 'time': now_iso(), 'result': '처리 중'}
        self.save()
        try:
            result = self.execute(plan, markets)
        except Exception as ex:
            result = f'중단: {type(ex).__name__}: {ex}'
        self.d['executed'][key]['result'] = result
        self.save()
        self.record({'event': '주문안 처리', 'id': plan[0], 'plan': plan[1], 'result': result})
        self.share(f'안건 {plan[0]} 「실행 주문안: {plan[1]}」 → {result}')

    def account_value(self):
        rows = upbit.accounts()
        coins = ['KRW-' + r['currency'] for r in rows if r['currency'] != 'KRW']
        listed = upbit.krw_markets()
        tick = upbit.tickers([m for m in coins if m in listed]) if coins else {}
        return sum(float(r['balance']) + float(r['locked']) for r in rows if r['currency'] == 'KRW') + sum(
            (float(r['balance']) + float(r['locked'])) * tick['KRW-' + r['currency']]['trade_price']
            for r in rows if 'KRW-' + r['currency'] in tick)

    def apply_limits(self, doc):
        found = latest_limits(doc)
        if not found or found[0] == self.d.get('limits_id'):
            return
        wanted = parse_limits(found[1])
        applied = throttle(self.cfg, wanted, self.account_value())
        self.cfg.update(applied)
        self.d['limits'] = {**self.d.get('limits', {}), **applied}
        self.d['limits_id'] = found[0]
        self.save()
        text = ', '.join(f'{k}={v:,}' for k, v in applied.items())
        capped = [k for k in wanted if applied[k] != wanted[k]]
        self.record({'event': '한도 변경', 'id': found[0], 'applied': applied, 'wanted': wanted})
        self.share(f'안건 {found[0]} 합의 한도 적용: {text}' + (f' (2배·계좌 총액 상한으로 조정: {", ".join(capped)})' if capped else ''))

    def poll_proposal(self, markets):
        f = self.folder.parent / 'proposal.json'
        if not f.exists():
            return
        prop = json.loads(f.read_text(encoding='utf-8'))
        if self.d.get('account_adoption'):
            created = dt.datetime.fromisoformat(prop['time']).timestamp()
            if not self.d['account_adoption']['epoch'] <= created <= time.time(): return
            if time.time() - created > self.cfg.get('proposal_max_age_seconds', 120): return
            # A fresh JSON write does not make an old strategy signal current.
            closed = prop.get('signal_closed_at')
            tf = prop.get('signal_tf_minutes')
            evaluated = prop.get('evaluated_at')
            now = time.time()
            if (type(tf) is not int or tf != self.p.tf or tf <= 0
                    or type(closed) not in (int, float) or not math.isfinite(closed)
                    or type(evaluated) not in (int, float) or not math.isfinite(evaluated)):
                return
            expected = int((now - 20) // (tf * 60)) * tf * 60
            if closed != expected or not closed + 20 <= evaluated <= now:
                return
        meta = None
        if rotation.enabled(self.p):
            state = self.d['rotation_state']
            if state.get('paused') or self.has_uncertain_orders(): return
            try:
                meta = rotation.validate(prop, self.p, time.time(),
                    max(state['floor_slot'], state.get('last_slot', -1)))
            except ValueError as error:
                rotation.append_audit(self.folder / 'rotation_audit.jsonl',
                    {'event': 'proposal_rejected', 'reason': str(error)})
                return
            state['last_slot'] = meta['slot']
            state['last_decision'] = meta['decision_id']
            state['claim_status'] = 'claimed_at_most_once_crash_may_skip'
            self.save()  # critical durable claim; diagnostics are optional
            meta = {**meta, 'markets': prop.get('rotation_markets', {})}
        text = prop['text'].split('실행 주문안:', 1)[-1].strip()
        key = ('R' + meta['decision_id']) if meta else 'P' + hashlib.sha256(f"{prop['time']}|{text}".encode()).hexdigest()[:15]
        if key in self.d['executed']:
            return
        label = f"봇 주문안 {prop['time'][5:16]}"
        self.d['executed'][key] = {'id': label, 'text': text, 'time': now_iso(), 'result': '처리 중'}
        self.save()
        self._rotation_meta = meta
        try:
            result = self.execute((label, text), markets,
                                  entry_context=prop.get('entry_context'), automatic=True,
                                  market_index=prop.get('market_index'))
        except Exception as ex:
            result = f'중단: {type(ex).__name__}: {ex}'
        self._rotation_meta = None
        if meta:
            self.d['rotation_state']['claim_status'] = 'processed'
            rotation.append_audit(self.folder / 'rotation_audit.jsonl',
                {'event': 'decision_processed', 'decision': meta, 'result': result})
        self.d['executed'][key]['result'] = result
        self.save()
        self.record({'event': '주문안 처리', 'id': label, 'plan': text, 'result': result})
        self.share(f'{label} (사용자 위임으로 자동 실행) 「{text}」 → {result}')

    def execute(self, plan, markets, entry_context=None, automatic=False, market_index=None):
        if self.has_uncertain_orders(): return '미확정 주문 대사 중: 새 주문안 보류'
        orders = parse(plan[1])
        protected = self.protected_markets()
        protected_orders = [m for _, m, _ in orders if m in protected]
        orders = [(side, m, amount) for side, m, amount in orders if m not in protected]
        regime_blocked = []
        if automatic and getattr(self.p, 'halt_new_buys', False):
            regime_blocked = [m for side, m, _ in orders if side == 'buy']
            orders = [o for o in orders if o[0] != 'buy']
            if regime_blocked:
                self.record({'event': '신규 매수 임시 중단으로 차단', 'markets': regime_blocked})
        elif automatic and getattr(self.p, 'regime_filter', False):
            # 수신 쪽에서도 장세 필터를 강제한다: 오래된 생성기나 필터 정보 누락 주문안이 신규 매수를 만들지 못하게 한다. 매도는 그대로.
            expires = getattr(self.p, 'regime_expires', 0.0)
            expired = bool(expires) and time.time() >= expires
            if expired or not (type(market_index) in (int, float) and math.isfinite(market_index) and market_index > 0):
                regime_blocked = [m for side, m, _ in orders if side == 'buy']
                orders = [o for o in orders if o[0] != 'buy']
                if regime_blocked:
                    self.record({'event': '장세 필터 매수 차단', 'markets': regime_blocked, 'market_index': market_index})
        sold_markets = {m for side, m, _ in orders if side == 'sell'}
        bought_markets = {m for side, m, _ in orders if side == 'buy'}
        if sold_markets & bought_markets:
            # Reject before the first account request or order; old generators cannot sell then rebuy.
            return '실행 거부: 동일 주문안 매도·매수 종목 충돌 ' + ', '.join(sorted(sold_markets & bought_markets))
        if not orders:
            if regime_blocked:
                return '장세 필터: 신규 매수 차단 ' + ', '.join(sorted(set(regime_blocked)))
            return ('사용자 보호 종목 주문 제외: ' + ', '.join(sorted(set(protected_orders)))) if protected_orders else '주문 없음 (매도 없음·매수 없음)'
        acc = {r['currency']: r for r in upbit.accounts()}
        tick = upbit.tickers([m for _, m, _ in orders])
        done = []
        sell_value = sum(float(acc.get(m[4:], {}).get('balance', 0)) * pct / 100 * tick[m]['trade_price']
                         for side, m, pct in orders if side == 'sell' and m in tick)
        if sell_value > self.cfg['max_sell_krw_per_plan']:
            return f'실행 안 함: 매도 예상 {sell_value:,.0f}원이 1회 한도 {self.cfg["max_sell_krw_per_plan"]:,}원 초과'
        for side, m, pct in orders:
            if side != 'sell':
                continue
            avail = float(acc.get(m[4:], {}).get('balance', 0))
            vol = int(avail * pct / 100 * 1e8) / 1e8
            if vol * tick[m]['trade_price'] < 5000:
                done.append(f'{m[4:]} 매도 건너뜀(5,000원 미만)')
                continue
            o = self._trade('ask', m, vol, plan[0])
            v, funds, fee = self._fill(o)
            done.append(f'{m[4:]} 매도 {v:g}개 → {funds - fee:,.0f}원 ({o["state"]})')
        self.save()
        for side, m, krw in orders:
            if side != 'buy':
                continue
            # Automatic buys always require source context. Manual plan buys also
            # require it while the chase cap is enabled; absent context is an
            # explicit skip, never an implicit bypass. Disabled-cap manual policy
            # and private low-level recovery/test callers retain their old contract.
            cap = self.p.chase_rs_cap
            strict = (automatic or type(cap) not in (int, float)
                      or not math.isfinite(cap) or cap > 0)
            context = entry_context.get(m) if isinstance(entry_context, dict) else None
            why = (self.buy_block(m, krw, markets, entry_context=context, validate_entry=True)
                   if strict else self.buy_block(m, krw, markets))
            if why:
                done.append(f'{m[4:]} 매수 안 함: {why}')
                continue
            o = (self._trade('bid', m, krw, plan[0], entry_context=context, validate_entry=True)
                 if strict else self._trade('bid', m, krw, plan[0]))
            v, funds, fee = self._fill(o)
            if v <= 0:
                done.append(f'{m[4:]} 매수 미체결 ({o["state"]})')
                continue
            stop = self.d['positions'][m]['stop']
            done.append(f'{m[4:]} 매수 {funds + fee:,.0f}원 · 손절가 {stop:,.4g}')
            self.save()
        return ' / '.join(done) or '실행할 주문 없음'

    def entry_chase_block(self, market, context, ask=_NO_ASK):
        """Validate source identity and enforce the receiver's existing cap; no new threshold."""
        if rotation.enabled(self.p):
            if (not isinstance(context, dict) or type(context.get('rank')) is not int
                    or not 1 <= context['rank'] <= self.p.M
                    or context.get('policy_hash') != rotation.policy_hash(self.p)):
                return '신규 매수 순위·유지 범위·정책 불일치: 매수 보류'
            why = rotation.fresh_reentry(self.d.get('last_sell', {}).get(market),
                                         self.p, context.get('signal_closed_at'))
            if why: return why
        if not isinstance(context, dict) or context.get('market') != market:
            return '매수 신호 출처 누락·종목 불일치: 매수 보류'
        close, anchor, rs = (context.get(k) for k in ('signal_close', 'anchor_close', 'rs'))
        if (any(type(v) not in (int, float) or not math.isfinite(v) for v in (close, anchor, rs))
                or close <= 0 or anchor <= 0 or rs <= 0):
            return '매수 신호 기준가격·상승률 무효: 매수 보류'
        if not math.isclose(close / anchor - 1, rs, rel_tol=1e-9, abs_tol=1e-12):
            return '매수 신호 가격과 상승률 불일치: 매수 보류'
        closed, tf, evaluated = (context.get(k) for k in ('signal_closed_at', 'signal_tf_minutes', 'evaluated_at'))
        now = time.time()
        if (type(tf) is not int or tf != self.p.tf or tf <= 0
                or any(type(v) not in (int, float) or not math.isfinite(v) for v in (closed, evaluated))):
            return '매수 신호 시각·주기 무효: 매수 보류'
        expected = int((now - 20) // (tf * 60)) * tf * 60
        if closed != expected or not closed + 20 <= evaluated <= now:
            return '매수 신호가 최신 마감 판단과 다름: 매수 보류'
        cap = self.p.chase_rs_cap
        if type(cap) not in (int, float) or not math.isfinite(cap):
            return '과열 상한 설정 무효: 매수 보류'
        if cap > 0 and rs >= cap:
            return '신호 시점부터 기존 과열 상한 이상: 매수 보류'
        if ask is not _NO_ASK:
            if type(ask) not in (int, float) or not math.isfinite(ask) or ask <= 0:
                return '매수 호가 무효: 매수 보류'
            # Price comparison avoids rounding an exact 15% boundary below the cap.
            if cap > 0 and ask >= anchor * (1 + cap):
                return f'실행 호가가 기존 과열 상한 {cap:.1%} 이상: 추격 매수 보류'
        return None

    def buy_block(self, m, krw, markets, entry_context=None, validate_entry=False):
        if self.is_protected(m): return '사용자 보호 종목: 자동·추가 매수 금지'
        c = self.cfg
        if self.d['halted']:
            return '손실 중단선 도달로 신규 매수 중단 상태'
        last = self.d.get('last_sell', {}).get(m)
        if last:
            # Re-read the current durable state, not the producer's pre-sale snapshot.
            if type(last.get('time')) not in (int, float) or not math.isfinite(last['time']):
                return '현재 매도 기록 시각 무효: 재진입 보류'
            elapsed = time.time() - last['time']
            if elapsed < 0:
                return '현재 매도 기록이 미래 시각: 재진입 보류'
            hours = self.p.reentry_profit_hours if last.get('profit') else self.p.reentry_loss_hours
            if 0 <= elapsed < min(hours, self.p.reentry_window_hours) * 3600:
                return '현재 매도 장부 기준 재진입 대기: ' + str(m)
        if self.has_uncertain_orders():
            return '미확정 주문 대사 중: 신규 매수 보류'
        if self.d.get('valuation_incomplete'):
            return '일부 보유분 가격 미확인: 신규 매수 보류'
        if m not in markets:
            return '원화 마켓에 없음'
        if markets[m].get('warning'):
            return '업비트 경보 종목'
        if krw < 5000:
            return '5,000원 미만'
        if krw > c['max_order_krw']:
            return f'1건 한도 {c["max_order_krw"]:,}원 초과'
        if self.d['bought_krw'] + krw * (1 + self.p.fee) > c['max_total_buy_krw']:
            return f'누적 매수 한도 {c["max_total_buy_krw"]:,}원 초과 (현재 {self.d["bought_krw"]:,.0f}원)'
        if validate_entry:
            why = self.entry_chase_block(m, entry_context)
            if why: return why
        krw_avail = float(next((r['balance'] for r in upbit.accounts() if r['currency'] == 'KRW'), 0))
        if krw * 1.0005 > krw_avail:
            return f'가용 원화 부족 ({krw_avail:,.0f}원)'
        ob = upbit.orderbooks([m])[m]
        if (ob['ask'] - ob['bid']) / ob['ask'] > c['max_spread']:
            return f'호가 차이 {((ob["ask"] - ob["bid"]) / ob["ask"]):.2%} > {c["max_spread"]:.0%}'
        if validate_entry:
            why = self.entry_chase_block(m, entry_context, ob.get('ask'))
            if why: return why
        if (last and last.get('profit') and self.p.reentry_dip_atr > 0
                and time.time() - last['time'] < self.p.reentry_window_hours * 3600):
            reentry_bars = upbit.completed_candles(m, self.p.tf, 100)
            atr = st.wilder_atr(reentry_bars)
            if atr is None or not math.isfinite(atr) or atr <= 0:
                return '재진입 판정 ATR 자료 부족'
            reentry_row = {'close': ob['ask'], 'atr': atr}
            if rotation.enabled(self.p) and reentry_bars:
                reentry_row['bar'] = reentry_bars[-1]['t']
            why = st.reentry_block(reentry_row, last, time.time(), self.p)
            if why: return why
        return ''

    def rebase_adopted_stops(self, prices):
        """인수 보유분의 손절가가 현재가보다 stop_max_pct 넘게 멀면 현재가 기준 상한으로 한 번 올린다(내리지 않음).
        실시간 보호 기준인 stop0에도 같은 상한을 한 번 적용한다(stop0_rebased)."""
        if not self.p.stop_max_pct > 0: return
        changed, changed0 = [], []
        for m, pos in self.d['positions'].items():
            px = prices.get(m)
            if self.is_protected(m): continue
            if type(px) not in (int, float) or not math.isfinite(px) or px <= 0: continue
            if not pos.get('stop_rebased'):
                new = st.cap_stop(px, pos['stop'], self.p)
                pos['stop_rebased'] = True
                if new > pos['stop']:
                    changed.append((m, pos['stop'], new))
                    pos['stop'] = new
            if 'stop0' in pos and not pos.get('stop0_rebased'):
                new = st.cap_stop(px, pos['stop0'], self.p)
                pos['stop0_rebased'] = True
                if new > pos['stop0']:
                    changed0.append((m, pos['stop0'], new))
                    pos['stop0'] = new
        if changed:
            for m, old, new in changed:
                self.record({'event': '손절가 상한 적용', 'market': m, 'old_stop': old, 'new_stop': new,
                             'rule': f'손절가는 현재가의 -{self.p.stop_max_pct:.0%}보다 멀 수 없음(결정 D-12)'})
        for m, old, new in changed0:
            self.record({'event': '실시간 손절가 상한 적용', 'market': m, 'old_stop0': old, 'new_stop0': new,
                         'rule': f'실시간 보호 손절(stop0)도 현재가의 -{self.p.stop_max_pct:.0%}보다 멀 수 없음(결정 D-12)'})
        if changed or changed0 or self.d['positions']: self.save()

    def stop_limit(self, pos):
        """EXIT-T1 대상은 초기 고정 손절만 실시간 가격으로 보호하고, 상향된 손절가는 확정 1분봉에서 평가한다.
        profit_guard_enabled이면 이익 보호선(profit_guard_stop)도 실시간 가격으로 함께 평가한다."""
        if self.cfg.get('trail_enabled') and pos.get('trail'):
            base = pos.get('stop0', pos['stop'])
        else:
            base = pos['stop']
        g = pos.get('profit_guard_stop') if self.cfg.get('profit_guard_enabled') else None
        return max(base, g) if type(g) in (int, float) and math.isfinite(g) else base

    def update_profit_guard(self, prices):
        """도입 후 관측한 고점이 평단 대비 arm_pct 이상이면 고점 이익의 (1-giveback)를 보존하는 보호선을 올린다(내리지 않음).
        수수료 바닥은 전략 수수료 추정치이며 실제 체결가를 보장하지 않는다."""
        if self.cfg.get('profit_guard_enabled') is not True: return
        a = self.cfg.get('profit_guard_arm_pct', 0.02)
        g = self.cfg.get('profit_guard_giveback', 0.4)
        if isinstance(a, bool) or isinstance(g, bool) or isinstance(self.p.fee, bool): return
        try: arm, give, fee = float(a), float(g), float(self.p.fee)
        except (TypeError, ValueError, OverflowError, AttributeError): return
        if not (math.isfinite(arm) and arm > 0 and math.isfinite(give) and 0 < give < 1
                and math.isfinite(fee) and fee >= 0): return
        ok = lambda x: type(x) in (int, float) and math.isfinite(x) and x > 0
        changed = False
        for m, pos in self.d['positions'].items():
            px, entry, q = prices.get(m), pos.get('entry'), pos.get('qty')
            if self.is_protected(m): continue
            if ok(q) and ok(pos.get('cost')): entry = pos['cost'] / q
            if not (ok(px) and ok(entry) and ok(q)): continue
            try: c = float(st.cost(m, self.p))
            except (TypeError, ValueError, OverflowError): continue
            floor = 1 + 2 * fee + c
            if not (math.isfinite(c) and c >= 0 and math.isfinite(floor)): continue
            old_peak = pos.get('profit_guard_peak') if pos.get('profit_guard_v') == 1 else None
            if old_peak is None:
                pos['profit_guard_v'] = 1
                changed = True
            peak = max(old_peak if ok(old_peak) else 0, entry, px)
            if peak != old_peak:
                pos['profit_guard_peak'] = peak
                changed = True
            if peak < entry * (1 + arm): continue
            old = pos.get('profit_guard_stop')
            old = old if ok(old) else 0
            new = max(old, entry * floor, entry + (peak - entry) * (1 - give))
            if new > old:
                if not old:
                    self.record({'event': '이익보호 시작', 'market': m, 'entry': entry, 'peak': peak, 'guard_stop': new})
                pos['profit_guard_stop'] = new
                changed = True
        if changed: self.save()

    def adopt_trail_init(self):
        """인수 보유분(entry 없음)에 추적 기준을 한 번 만든다. E=거래소 평균 매수가(cost/qty), A=현재 15분 ATR, 고점=편입 이후 확정 15분봉 고점.
        초기 고정 손절 stop0는 현재 손절가 그대로 쓴다. 진입 시점을 모르므로 편입 시각 이후 봉만 쓰고 R은 3×ATR15로 정의한다(D-22)."""
        done = False
        for m, pos in self.d['positions'].items():
            if self.is_protected(m): continue
            if 'entry' in pos or pos.get('adopted_trail_failed') or pos['qty'] <= 0: continue
            try:
                bars = upbit.completed_candles(m, 15, 100)
                atr = st.wilder_atr(bars)
            except Exception:
                continue
            entry = pos['cost'] / pos['qty']
            if not atr or not math.isfinite(atr) or atr <= 0 or entry <= 0:
                pos['adopted_trail_failed'] = True
                continue
            after = [b['h'] for b in bars if st.bar_after_entry(b, pos['time'])]
            pos.update(entry=entry, atr=atr, stop0=pos['stop'], trail=True, adopted_trail=True,
                       peak=max([entry] + after))
            self.record({'event': '인수 보유분 추적 기준 생성', 'market': m, 'entry': entry, 'atr15': atr, 'peak': pos['peak'],
                         'rule': '거래소 평균 매수가=E, R=3×ATR15, 1R부터 추적(결정 D-22)'})
            done = True
        return done

    def trail_check(self):
        """EXIT-T1(실거래, 기본 꺼짐). 확정 1분봉 종가로 봉마다 한 번 평가. 봇이 새로 산 포지션만 대상."""
        from dataclasses import replace
        if not self.cfg['enabled'] or not self.cfg.get('trail_enabled'): return
        if not self.recover_all_pending(): return
        p = replace(self.p, trail_enabled=True)
        exits, parts, changed = [], [], False
        if self.cfg.get('trail_adopted') and self.adopt_trail_init(): changed = True
        for m, pos in list(self.d['positions'].items()):
            if self.is_protected(m): continue
            if not pos.get('trail') or pos['qty'] <= 0 or 'entry' not in pos: continue
            p_pos = replace(p, trail_start_r=float(self.cfg.get('adopted_trail_start_r', 1.0))) if pos.get('adopted_trail') else p
            try: bars = upbit.completed_candles(m, 1, 3)
            except Exception: continue
            bar = bars[-1] if bars else None
            if not bar or pos.get('last_bar') == bar['t'] or not st.bar_after_entry(bar, pos['time']): continue
            pos['last_bar'] = bar['t']
            changed = True
            why = st.ratchet(m, pos, bar['c'], p_pos)
            if why and not self.market_has_uncertain_order(m): exits.append((m, pos, why, bar['c']))
            elif not why and self.cfg.get('partial_enabled') and not pos.get('part_done') and pos['atr'] > 0:
                if (bar['c'] - pos['entry']) / (p_pos.k * pos['atr']) >= float(self.cfg.get('partial_r', 2.0)):
                    parts.append((m, pos, bar['c']))
        if changed: self.save()
        if not exits and not parts: return
        balances = {r['currency']: r for r in upbit.accounts()}
        for m, pos, why, price in exits:
            volume = min(pos['qty'], float(balances.get(m[4:], {}).get('balance', 0)))
            if volume * price < 5000: continue
            try:
                o = self._trade('ask', m, volume, why, risk=True)
                v, funds, fee = self._fill(o)
                self.share(f'{m[4:]} {why} 매도 {funds - fee:,.0f}원 (실현 누적 {self.d["realized"]:+,.0f}원)')
            except SessionLost:
                raise
            except Exception as error:
                self.record({'event': '종목 익절 청산 대기', 'market': m,
                             'reason': type(error).__name__ + ': ' + str(error)[:250]})
        for m, pos, price in parts:
            if self.market_has_uncertain_order(m) or m not in self.d['positions']: continue
            avail = float(balances.get(m[4:], {}).get('balance', 0))
            vol = int(min(pos['qty'], avail) * float(self.cfg.get('partial_frac', 1 / 3)) * 1e8) / 1e8
            if vol * price < 5000 or (pos['qty'] - vol) * price < 5000:
                pos['part_done'] = True  # 너무 작아 나눠 팔 수 없다: 다시 시도하지 않는다
                self.save()
                continue
            try:
                o = self._trade('ask', m, vol, '부분 익절', risk=True)
                v, funds, fee = self._fill(o)
                if m in self.d['positions']: self.d['positions'][m]['part_done'] = True
                self.save()
                self.share(f'{m[4:]} 부분 익절 {funds - fee:,.0f}원 (수익 {(price / pos["entry"] - 1):+.1%} 구간, 실현 누적 {self.d["realized"]:+,.0f}원)')
            except SessionLost:
                raise
            except Exception as error:
                self.record({'event': '부분 익절 대기', 'market': m, 'reason': type(error).__name__ + ': ' + str(error)[:250]})

    def liquidate_position(self, market, prices, reason='수동 강제 청산'):
        """사용자가 화면에서 확인한 종목 한 개를 전량 시장가 매도한다. (성공 여부, 안내문)을 돌려준다.

        봇이 관리하는 장부 종목만 허용하고 손절과 같은 위험 청산 경로(_trade risk=True)를 쓴다.
        그래서 주문 식별자 선기록·실행권 검사·미확정 주문 대사·장부 반영이 손절과 동일하다.
        """
        if self.is_protected(market): return False, '사용자 보호 종목: 해제 전에는 봇 화면 주문도 금지'
        if not self.cfg['enabled']: return False, '실거래가 꺼져 있음'
        pos = self.d['positions'].get(market)
        if not pos or not pos.get('qty', 0) > 0: return False, '봇이 관리하는 보유 종목이 아님'
        price = prices.get(market)
        if type(price) not in (int, float) or not math.isfinite(price) or price <= 0:
            return False, '현재가를 확인할 수 없어 청산 보류'
        if self.market_has_uncertain_order(market): return False, '미확정 주문 대사 중: 청산 보류'
        balances = {r['currency']: r for r in upbit.accounts()}
        volume = min(pos['qty'], float(balances.get(market[4:], {}).get('balance', 0)))
        if volume * price < 5000:
            return False, f'평가액 {volume * price:,.0f}원: 최소 주문금액 5,000원 미만이라 청산할 수 없음'
        self.record({'event': '수동 청산 요청', 'market': market, 'reason': reason})
        try:
            o = self._trade('ask', market, volume, reason, risk=True)
        except SessionLost:
            raise
        except Exception as error:
            self.record({'event': '수동 청산 실패', 'market': market, 'reason': type(error).__name__ + ': ' + str(error)[:250]})
            return False, f'청산 실패: {type(error).__name__}: {str(error)[:120]}'
        v, funds, fee = self._fill(o)
        if v <= 0:
            return False, f'매도 미체결 ({o.get("state")}): 보유는 그대로입니다'
        left = self.d['positions'].get(market, {}).get('qty', 0)
        msg = f'{market[4:]} 청산 {v:,.8g}개 → {funds - fee:,.0f}원' + (f' (일부 체결, 남은 수량 {left:,.8g})' if left > 0 else '')
        self.share(f'{msg} (수동 강제 청산, 실현 누적 {self.d["realized"]:+,.0f}원)')
        return True, msg

    def add_to_position(self, market, krw, markets, reason='수동 추가 매수'):
        """사용자가 화면에서 확인한 금액만큼 이미 관리 중인 종목을 시장가로 더 산다. (성공 여부, 안내문)을 돌려준다.

        자동 매수와 같은 검사(buy_block: 손실 중단·미확정 주문·한도·가용 원화·호가 차이)를 모두 통과해야 한다.
        기존 규칙상 추가 매수한 종목은 추적 손절·이익 보호 대상에서 빠진다(_apply_pending, 안건 27 미합의).
        """
        if self.is_protected(market): return False, '사용자 보호 종목: 해제 전에는 봇 화면 주문도 금지'
        if not self.cfg['enabled']: return False, '실거래가 꺼져 있음'
        if market not in self.d['positions']: return False, '봇이 관리하는 보유 종목이 아님: 새 종목 매수는 지원하지 않음'
        cap = min(self.cfg.get('manual_buy_max_krw', 100000), self.cfg['max_order_krw'])
        if type(krw) is not int or krw < 5000: return False, '매수 금액은 5,000원 이상의 정수여야 함'
        if krw > cap: return False, f'1회 추가 매수 상한 {cap:,}원 초과'
        why = self.buy_block(market, krw, markets)
        if why: return False, f'매수 보류: {why}'
        self.record({'event': '수동 추가 매수 요청', 'market': market, 'krw': krw})
        try:
            o = self._trade('bid', market, krw, reason)
        except SessionLost:
            raise
        except Exception as error:
            self.record({'event': '수동 추가 매수 실패', 'market': market, 'reason': type(error).__name__ + ': ' + str(error)[:250]})
            return False, f'매수 실패: {type(error).__name__}: {str(error)[:120]}'
        v, funds, fee = self._fill(o)
        if v <= 0:
            return False, f'매수 미체결 ({o.get("state")}): 보유는 그대로입니다'
        pos = self.d['positions'][market]
        msg = f'{market[4:]} 추가 매수 {funds + fee:,.0f}원 · 손절가 {pos["stop"]:,.4g} · 보유 평가 매수금액 {pos["cost"]:,.0f}원'
        self.share(f'{msg} (수동 추가 매수)')
        return True, msg

    def stop_check(self, prices):
        """Manage explicitly delegated positions, retaining partial fills and unresolved orders."""
        if not self.cfg['enabled']: return
        self.recover_all_pending()
        self.rebase_adopted_stops(prices)
        self.update_profit_guard(prices)
        if rotation.enabled(self.p):
            for m, pos in self.d['positions'].items():
                if not self.is_protected(m):
                    rotation.observe(pos, prices.get(m), time.time())
            # Optional observations persist with the next normal ledger save.
            # Do not add a disk write in front of urgent protective orders.
        candidates = [(m, pos) for m, pos in list(self.d['positions'].items())
                      if not self.is_protected(m) and m in prices and type(prices[m]) in (int, float) and math.isfinite(prices[m])
                      and prices[m] > 0 and prices[m] <= self.stop_limit(pos) and pos['qty'] > 0
                      and not self.market_has_uncertain_order(m)]
        if not candidates: return
        balances = {r['currency']: r for r in upbit.accounts()}
        for m, pos in candidates:
            available = float(balances.get(m[4:], {}).get('balance', 0))
            volume = min(pos['qty'], available)
            if volume * prices[m] < 5000: continue
            g = pos.get('profit_guard_stop') if self.cfg.get('profit_guard_enabled') else None
            why = '이익보호' if g is not None and self.stop_limit(pos) == g else '손절'
            try:
                o = self._trade('ask', m, volume, why, risk=True)
                v, funds, fee = self._fill(o)
                self.share(f'{m[4:]} {why} 매도 {funds - fee:,.0f}원 (실현 누적 {self.d["realized"]:+,.0f}원)')
            except SessionLost:
                raise
            except Exception as error:
                # One market's rejected/uncertain order must not suppress other owned-market exits.
                self.record({'event': '종목 청산 대기', 'market': m,
                             'reason': type(error).__name__ + ': ' + str(error)[:250]})

def boundary(ts, period):
    return int(ts // period * period)


def utc(ts):
    return dt.datetime.fromtimestamp(ts, dt.timezone.utc)


def ranking_rows(ranking, held, warned, p, esc=html.escape):
    """순위표 행. 바탕색: 보유=하늘색, 경보=붉은 파스텔, 투자 대상(매수 후보)=연두, 관찰=연노랑. '구분' 칸에도 적는다.
    매수 후보 = 현재 매수 규칙(상위 entry_top 안, 상승률 0 초과 chase_rs_cap 미만, 보유·경보 아님). 관찰 = 5위(유지 순위) 안 정상 종목."""
    out, want, scan, picked = [], p.entry_want or p.entry_top, p.entry_scan or p.entry_top, 0
    for r in ranking:
        m, rs, rank = r['market'], r['rs'], r['rank']
        if m in warned:
            cls, tag = 'rk-alert', '보유·경보' if m in held else '경보'
        elif m in held:
            cls, tag = 'rk-held', '보유'
        elif rank <= scan and rs >= p.chase_rs_cap > 0:
            cls, tag = 'rk-hot', '과열(매수 보류)'
        elif rank <= scan and rs > 0 and picked < want:
            cls, tag, picked = 'rk-target', '투자 대상(매수 후보)', picked + 1
        elif rank <= p.M and rs > 0:
            cls, tag = 'rk-watch', '관찰'
        else:
            cls, tag = '', ''
        out.append(f"<tr class='{cls}'><td class=n>{rank}</td><td>{esc(m[4:])}</td>"
                   f"<td class='n {'up' if rs >= 0 else 'dn'}'>{rs * 100:+.2f}%</td><td>{tag}</td></tr>")
    return ''.join(out)

    def align_timeframe(self, now):
        """Preserve holdings on a timeframe change and wait for the next new close."""
        if type(self.p.tf) is not int or self.p.tf not in {1, 3, 5, 10, 15, 30, 60, 240}:
            raise ValueError('지원하지 않는 판단 분봉입니다.')
        if type(self.p.stop_seconds) is not int or self.p.stop_seconds <= 0:
            raise ValueError('손절 확인 간격은 양의 정수 초여야 합니다.')
        for lg in self.ledgers.values():
            if lg.d.get('timeframe_minutes') == self.p.tf:
                continue
            lg.d['last_4h'] = boundary(now - SETTLE, self.p.tf * 60)
            lg.d['timeframe_minutes'] = self.p.tf
            lg.event(f'판단 주기 {self.p.tf}분 적용: 기존 수량·손절가 보존, 지난 신호 소급 처리 없음')
            lg.save()

    def init_t0(self):
        rows = upbit.accounts()
        krw = sum(float(r['balance']) + float(r['locked']) for r in rows if r['currency'] == 'KRW')
        listed = upbit.krw_markets()
        coins = {f"KRW-{r['currency']}": float(r['balance']) + float(r['locked']) for r in rows if r['currency'] != 'KRW'}
        priced = [m for m in coins if m in listed]
        tick = upbit.tickers(priced) if priced else {}
        basket = {m: q for m, q in coins.items() if m in tick}
        unpriced = sorted(m for m in coins if m not in tick)
        E = krw + sum(q * tick[m]['trade_price'] for m, q in basket.items())
        f = self.cfg.get('allocation_fraction', 0.10)
        if not self.ledgers['P10'].ready:
            self.ledgers['P10'].start(E, f, basket, krw, unpriced)
        if not self.ledgers['ALL'].ready:
            self.ledgers['ALL'].start(E, 1.0, basket, krw, unpriced,
                                      label='단독 전략 실험(계좌 평가액만큼 가상 현금, 실제 계좌 인수 아님)')
        if unpriced:
            self.log('평가 가격이 없는 보유자산은 E에서 제외: ' + ', '.join(unpriced))

    # ---------- 판단 ----------
    def candidates(self):
        self.markets = upbit.krw_markets()
        tick = upbit.tickers(self.markets)
        pool = [m for m in tick if m not in self.p.exclude]
        pool.sort(key=lambda m: -tick[m]['acc_trade_price_24h'])
        return pool[:self.p.universe], tick

    def fetch_bars(self, markets, end):
        out = {}
        for m in markets:
            bars = upbit.completed_candles(m, self.p.tf, 200)
            # 방금 마감된 봉이 있어야 판단에 쓴다 (준비 자료 부족이면 제외)
            if bars and dt.datetime.fromisoformat(bars[-1]['t']).timestamp() + self.p.tf * 60 == end:
                out[m] = bars
            time.sleep(0.12)
        return out

    def rebalance(self, end):
        universe, tick = self.candidates()
        held = [m for lg in self.ledgers.values() for m in lg.d['positions']]
        real = list(self.real.get('coins', {}))
        bars = self.fetch_bars(list(dict.fromkeys(universe + held + real)), end)
        uni = set(universe)
        # 신규 순위는 거래대금 대상 안에서만 계산한다. 대상 밖 보유는 순위를 차지하지 않는다
        ranked = st.rank({m: b for m, b in bars.items() if m in uni}, self.p)
        self.rs_all = {r['market']: (r['rank'], r['rs']) for r in ranked}
        for m in bars:
            if m not in uni:
                self.rs_all[m] = (None, st.relative_strength(bars[m], self.p.L))
        entry_ok = {m for m in universe if m in bars and not self.markets.get(m, {}).get('warning')}
        failed = []
        for name, lg in self.ledgers.items():
            if (lg.d['last_4h'] or 0) >= end:
                continue  # 이 마감은 이미 처리했다
            try:
                self.apply(name, lg, ranked, entry_ok, end, bars, uni)
            except Exception as ex:
                failed.append(f'{name}: {type(ex).__name__}: {ex}')
        top = ', '.join(f"{r['market'][4:]}{r['rs']:+.1%}" for r in ranked[:5])
        self.log(f'{self.p.tf}분 판단 완료 · 상위: {top}')
        if failed:
            raise RuntimeError('장부 처리 실패, 다음 주기에 그 장부만 재시도: ' + '; '.join(failed))

    def apply(self, name, lg, ranked, entry_ok, end, bars, uni):
        d = lg.d
        if rotation.enabled(self.p):
            floor = d.setdefault('rotation_first_slot', rotation.slot(time.time()))
            prior = max(floor, d.get('rotation_last_slot', floor))
            d['ranking'] = ranked[:15]
            d['ranking_time'] = utc(end).astimezone(KST).isoformat(timespec='minutes')
            if end % rotation.HOUR or end <= prior:
                d['last_4h'] = end
                lg.save()
                return
            d['rotation_last_slot'] = end
            d['last_4h'] = end
            lg.save()  # claim before any paper fill; crash skips this slot
        held = list(d['positions'])
        now = time.time()
        d['banned'] = {m: t for m, t in d['banned'].items() if t > now}
        sells, buys, unknown = st.plan(ranked, held, set(d['banned']), entry_ok, self.p)
        if rotation.enabled(self.p):
            index = st.market_index({m:b for m,b in bars.items() if m in uni}, self.p.regime_bars, self.p.regime_min_markets) if self.p.regime_filter else None
            expired = self.p.regime_filter and self.p.regime_expires and now >= self.p.regime_expires
            if self.p.halt_new_buys or expired or (self.p.regime_filter and (type(index) not in (int,float) or not math.isfinite(index) or index <= 0)):
                buys = []
        d['ranking'] = ranked[:15]
        d['ranking_time'] = utc(end).astimezone(KST).isoformat(timespec='minutes')
        log = lambda msg: (lg.event(msg), print(f'[{now_iso()}] {name} {msg}', flush=True))
        for m in unknown:
            why = '거래대금 상위 대상 밖' if m in bars and m not in uni else '마감 봉 자료 없음'
            log(f'{m}: {why} · 순위 판단 보류하고 유지 (누락·이탈 처리 17번 미합의)')
        books = upbit.orderbooks([m for m, _ in sells] + buys) if sells or buys else {}
        for m, reason in sells:
            r = lg.sell(m, books[m]['bid'], reason, self.p)
            log(f"{m} {reason} 매도 {r['amount']:,}원 (손익 {r['pnl']:+,}원, {r['pnl_pct']:+.2f}%)")
        atr = {r['market']: r['atr'] for r in ranked}
        for m in buys:
            prices = self.prices(lg)
            eq, inv = lg.equity(prices), lg.invested(prices)
            qty_fn = lambda P, S: st.size(eq, inv, d['cash'], P, S, self.p)
            r = lg.buy(m, books[m]['ask'], atr[m], qty_fn, self.p)
            if r:
                log(f"{m} 매수 {r['amount']:,}원 · 손절가 {r['stop']:,.4g}")
            else:
                d['skipped'] = d.get('skipped', 0) + 1
                log(f'{m} 매수 건너뜀: 계산된 주문이 최소주문 5,000원 미달')
        d['signals'] = d.get('signals', 0) + len(buys)
        d['last_4h'] = end
        lg.save()

    def stop_check(self, lg, live):

    def prices(self, lg=None):
        """최신 체결가. 응답에 빠진 종목은 마지막 관측값을 쓰고 price_missing에 기록한다."""
        ms = list(dict.fromkeys([m for x in self.ledgers.values() for m in x.d['positions']]
                                + list(self.ledger.d['H']['basket']) + list(self.real.get('coins', {}))
                                + list(self.live.d['positions'])))
        fresh = upbit.tickers(ms) if ms else {}
        now = time.time()
        for m, t in fresh.items():
            self.price_cache[m] = (t['trade_price'], now)
        self.price_missing = [m for m in ms if m not in fresh]
        return {m: v[0] for m, v in self.price_cache.items()}

            else:

        self.real = {'time': now_iso(), 'krw': krw, 'coins': coins, 'unpriced': unpriced}

    def real_block(self, prices):

    def run(self):
        subprocess.Popen([sys.executable, str(ROOT / 'account_sync.py')],
                         cwd=str(ROOT), creationflags=subprocess.CREATE_NO_WINDOW)
        if not all(lg.ready for lg in self.ledgers.values()):
            self.init_t0()
        self.align_timeframe(time.time())
        self.log('봇 실행 시작 (실거래 활성 · 기존 보유분 위임 관리 포함)' if self.live.cfg['enabled'] else '봇 실행 시작 (모의 운영)')
        last_snap = last_stop = last_live = last_display = last_trail = last_manual_scan = 0
        stop_iv = min(self.p.stop_seconds, 10) if self.live.cfg['enabled'] and self.live.cfg.get('profit_guard_enabled') is True else self.p.stop_seconds
        self.manual_server = None
        if self.live.cfg['enabled'] and (self.live.cfg.get('manual_ui') or self.live.cfg.get('protection_ui')):
            kinds=set()
            if self.live.cfg.get('manual_ui'): kinds.update(('buy','liquidate'))
            if self.live.cfg.get('protection_ui'): kinds.update(('protect','unprotect'))
            srv = manual_orders.ManualOrderServer(self.live.cfg.get('manual_port', manual_orders.DEFAULT_PORT),allowed_kinds=kinds)
            started=srv.start()
            if not started and self.live.cfg.get('protection_ui'):
                srv=manual_orders.ManualOrderServer(0,allowed_kinds=kinds)
                started=srv.start()
            if started:
                self.manual_server = srv
                self.log(f'화면 강제 청산 요청 서버 시작 (127.0.0.1:{srv.port}, 토큰 인증)')
            else:
                self.log(f'화면 강제 청산 서버를 열지 못함(포트 {srv.port} 사용 중): 청산 버튼 비활성')
        while not (STOP.exists() or STOP_ASCII.exists()):
            guard = getattr(self, 'session_guard', None)
            if guard and guard.lost.is_set():
                self.log('실행 연결 종료: ' + guard.reason)
                break
            now = time.time()
            try:
                if self.live.cfg.get('auto_manual_protection') and now-last_manual_scan>=15:
                    try: self.live.scan_manual_orders()
                    except Exception as ex: self.log('직접 주문 보호 확인 대기: '+type(ex).__name__)
                    last_manual_scan=now
                self.handle_manual_requests()
                if getattr(self,'_protection_ui_dirty',False):
                    last_display=0; self._protection_ui_dirty=False
                TF = self.p.tf * 60
                end4 = boundary(now - SETTLE, TF)
                held = list({m for lg in self.ledgers.values() for m in lg.d['positions']} | set(self.live.d['positions']))
                if held and now - last_stop >= stop_iv:
                    live = {m: t['trade_price'] for m, t in upbit.tickers(held).items()}
                    self.live.stop_check(live)  # 실시간 손절·이익보호를 1분봉 순차 조회(trail_check)보다 먼저
                    if now - last_trail >= self.p.stop_seconds:
                        self.live.trail_check()
                        last_trail = now
                    self.live.check_resume_request(live)
                    self.live.check_loss(live)
                    for lg in self.ledgers.values():
                        self.stop_check(lg, live)
                    last_stop = now
                if now - last_display >= 30:
                    self.refresh_display(end4 + TF + SETTLE)
                    last_display = now
                if now - last_live >= 30:
                    self.live.poll(self.markets or upbit.krw_markets())
                    last_live = now
                    self.refresh_display(end4 + TF + SETTLE)
                    last_display = now
                for lg in self.ledgers.values():
                    if not lg.d['last_4h']:  # 새 장부는 이미 지난 봉으로 매매하지 않는다
                        lg.d['last_4h'] = end4
                        lg.save()
                if any((lg.d['last_4h'] or 0) < end4 for lg in self.ledgers.values()):
                    self.rebalance(end4)
                if now - last_snap >= 300:

"""Reviewed rotation envelope, evidence, and research-only confirmation helpers.

No network, credentials, order submission, process, or independent collector.
Critical claim/pending state remains in Live's existing durable ledger.
"""
from dataclasses import asdict
import copy
import datetime as dt
import hashlib
import json
import math
from pathlib import Path
import re
import time

VERSION = 'rotation-v1'
HOUR = 3600
SETTLE = 20
MARKET = re.compile(r'^KRW-[A-Z0-9]{1,20}$')


def finite(x):
    return type(x) in (int, float) and math.isfinite(x)


def enabled(p):
    return getattr(p, 'rotation_enabled', False) is True


def saved_loss_budget(cfg, ledger):
    """Pure read: never fetch/save inside partially applied fill accounting."""
    amount, pct, base = cfg.get('loss_stop_krw'), cfg.get('loss_stop_pct', 0), ledger.get('risk_base_equity')
    if not finite(amount) or amount <= 0 or not finite(pct):
        return None
    if pct > 0 and finite(base) and base > 0:
        return min(amount, base * pct)
    return amount  # Same fixed-limit fallback when a stored percent basis is absent.


def slot(now):
    if not finite(now) or now < 0:
        raise ValueError('Invalid rotation clock')
    return int(now // HOUR) * HOUR


def policy_hash(p):
    # Params contains public strategy settings only, never the .env or live tokens.
    return hashlib.sha256(json.dumps(asdict(p), sort_keys=True, separators=(',', ':')).encode()).hexdigest()


def decision_id(closed, digest):
    return hashlib.sha256(f'{VERSION}:{closed}:{digest}'.encode()).hexdigest()


def intent_id(closed, market, side, digest):
    if not MARKET.fullmatch(market) or side not in ('bid', 'ask'):
        raise ValueError('Invalid rotation intent')
    return hashlib.sha256(f'{VERSION}:{closed}:{market}:{side}:{digest}'.encode()).hexdigest()


def envelope(p, closed, now):
    if (type(closed) is not int or closed % HOUR or closed != slot(now)
            or not closed + SETTLE <= now < closed + p.tf * 60 + SETTLE):
        raise ValueError('Rotation needs the current hourly completed exchange signal')
    digest = policy_hash(p)
    return {'version': VERSION, 'slot': closed, 'policy_hash': digest,
            'decision_id': decision_id(closed, digest)}


def validate(prop, p, now, floor_slot):
    value = prop.get('rotation')
    if not isinstance(value, dict):
        raise ValueError('Missing rotation envelope; legacy proposal disabled')
    closed = prop.get('signal_closed_at')
    expected = envelope(p, closed, now)
    if value != expected or closed <= floor_slot:
        raise ValueError('Old, changed, or already baselined rotation signal')
    if prop.get('signal_tf_minutes') != p.tf:
        raise ValueError('Rotation source timeframe mismatch')
    return expected


def fresh_reentry(last, p, closed):
    if not last:
        return None
    stamp = last.get('time')
    hours = p.reentry_profit_hours if last.get('profit') else p.reentry_loss_hours
    if not finite(stamp) or not finite(closed) or not finite(hours) or hours < 0:
        return '재진입 기록·신호 무효: 매수 보류'
    wait = min(hours, p.reentry_window_hours) * 3600
    if closed <= stamp + wait:
        return '재진입 대기 만료 이후 새 완료봉 필요: 매수 보류'
    return None


def row_close(row, tf):
    try:
        value = dt.datetime.fromisoformat(row['bar'])
        if value.tzinfo is None:
            return None
        return value.timestamp() + tf * 60
    except (KeyError, ValueError, TypeError):
        return None


def rank_exit(state, market, rank, closed, keep, confirmations=1):
    """Research knob only: production patch uses 1, not an invented 2-bar delay."""
    if type(confirmations) is not int or confirmations not in (1, 2):
        raise ValueError('Invalid rank confirmation research setting')
    if type(rank) is not int or rank <= 0 or not finite(closed):
        return False  # Missing data is not a confirmed boundary exit.
    if rank <= keep:
        state.pop(market, None)
        return False
    old = state.get(market, {})
    count = old.get('count', 0)
    if closed > old.get('closed', -1):
        count += 1
        state[market] = {'closed': closed, 'count': count}
    return count >= confirmations


def observe(pos, price, now):
    """Observed ticks only, bounded by entry/basis revision; never backfill history."""
    if not finite(price) or price <= 0 or not finite(pos.get('qty')) or pos['qty'] <= 0:
        return False
    if not finite(pos.get('cost')) or not finite(now):
        return False
    ref = pos['cost'] / pos['qty']
    if not finite(ref) or ref <= 0:
        return False
    value = pos.get('rotation_observed')
    if (not isinstance(value, dict) or not finite(value.get('reference'))
            or not math.isclose(value['reference'], ref, rel_tol=1e-10)
            or not all(finite(value.get(k)) for k in ('high', 'low', 'since'))
            or type(value.get('samples')) is not int or value['samples'] < 0):
        value = {'since': now, 'reference': ref, 'high': ref, 'low': ref, 'samples': 0}
        pos['rotation_observed'] = value
    value.update(high=max(value['high'], price), low=min(value['low'], price),
                 last_price=price, last_at=now, samples=value['samples'] + 1)
    value.update(mfe_pct=(value['high'] / ref - 1) * 100,
                 mae_pct=(value['low'] / ref - 1) * 100,
                 coverage='observed_ticks_only_buy_fee_in_basis_no_gap_backfill')
    return True


def trade_evidence(pos, reason, now, meta=None):
    keys = ('stop', 'stop0', 'profit_guard_stop', 'qty', 'cost', 'rotation_observed', 'rotation_initial_risk')
    out = {'version': VERSION, 'intent_created_at': now, 'exit_reason': str(reason)[:100],
           'wire_submission_time': 'not_instrumented_use_exchange_order_time_if_provided'}
    if pos:
        out['position_at_submit'] = copy.deepcopy({k: pos[k] for k in keys if k in pos})
    if isinstance(meta, dict):
        out['decision'] = copy.deepcopy(meta)
    return out


def append_audit(path, event, max_bytes=4*1024*1024):
    """Optional diagnostics must not block risk orders; <=3 bounded log segments."""
    try:
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        payload = json.dumps({'observed_at': time.time(), **event}, ensure_ascii=False,
                             allow_nan=False, separators=(',', ':'))
        if len(payload.encode('utf-8')) > 64000:
            return False
        if path.exists() and path.stat().st_size + len(payload.encode('utf-8')) > max_bytes:
            first = path.with_name(path.name + '.1')
            second = path.with_name(path.name + '.2')
            if first.exists():
                first.replace(second)
            path.replace(first)
        with path.open('a', encoding='utf-8') as out:
            out.write(payload + '\n')
        return True
    except (OSError, ValueError, TypeError, OverflowError):
        return False

import json

class SlotDue(unittest.TestCase):
    def test_waits_for_delay_after_the_top_of_the_hour(self):
        self.assertIsNone(sh.slot_due(BASE + 10, BASE - H))
        self.assertIsNone(sh.slot_due(BASE + sh.SLOT_DELAY - 1, BASE - H))
        self.assertEqual(sh.slot_due(BASE + sh.SLOT_DELAY, BASE - H), BASE)

    def test_same_slot_is_never_due_twice(self):
        self.assertIsNone(sh.slot_due(BASE + 120, BASE))

    def test_late_slot_is_skipped_not_replayed(self):
        # 재시작이 슬롯 시작 뒤 허용 지연보다 늦으면 놓친 슬롯을 소급 실행하지 않는다.
        self.assertEqual(sh.slot_due(BASE + sh.SLOT_MAX_LATE, BASE - H), BASE)
        self.assertIsNone(sh.slot_due(BASE + sh.SLOT_MAX_LATE + 1, BASE - H))
        self.assertIsNone(sh.slot_due(BASE + 1200, BASE - H))

    def test_restart_in_the_middle_of_an_hour_creates_no_plan(self):
        for minutes in (5, 20, 37, 49, 59):
            now = BASE + minutes * 60
            last = sh.current_slot(now)  # 첫 가동: 현재 슬롯을 처리한 것으로 간주
            self.assertIsNone(sh.slot_due(now, last))
            self.assertIsNone(sh.slot_due(now + 30, last))


class SlotPersistence(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.path = Path(self.tmp.name) / 'plan_slot.json'

    def test_missing_or_corrupt_file_counts_as_current_slot_done(self):
        now = BASE + 200
        self.assertEqual(sh.load_last_slot(now, self.path), BASE)
        for text in ('', '{', '[]', '{"slot": "x"}', '{"slot": -5}', '{"slot": NaN}'):
            self.path.write_text(text, encoding='utf-8')
            self.assertEqual(sh.load_last_slot(now, self.path), BASE, text)
        self.assertIsNone(sh.slot_due(now, sh.load_last_slot(now, self.path)))

    def test_claim_is_durable_and_blocks_a_second_plan_after_restart(self):
        now = BASE + 120
        last = BASE - H
        slot = sh.slot_due(now, last)
        self.assertEqual(slot, BASE)
        sh.claim_slot(slot, self.path)
        self.assertEqual(json.loads(self.path.read_text(encoding='utf-8'))['slot'], BASE)
        # 프로세스가 죽었다 다시 떠도(메모리 상태 소실) 기록으로 같은 슬롯은 다시 만들어지지 않는다.
        restarted_last = sh.load_last_slot(now + 30, self.path)
        self.assertIsNone(sh.slot_due(now + 30, restarted_last))
        self.assertEqual(sh.slot_due(BASE + H + 120, restarted_last), BASE + H)

    def test_claim_happens_before_generation_so_a_crash_cannot_duplicate(self):
        # 생성기 흐름과 같은 순서를 흉내: claim -> (장애) -> 재시작 -> 같은 슬롯 재생성 시도
        slot = sh.slot_due(BASE + 100, BASE - H)
        sh.claim_slot(slot, self.path)
        generated = []
        try:
            raise RuntimeError('생성 중 장애')
        except RuntimeError:
            pass
        last = sh.load_last_slot(BASE + 130, self.path)
        if sh.slot_due(BASE + 130, last) is not None:
            generated.append('again')
        self.assertEqual(generated, [])


class Hardening(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.path = Path(self.tmp.name) / 'plan_slot.json'

    def test_restart_inside_the_late_window_waits_for_the_next_slot(self):
        sh.claim_slot(BASE - H, self.path)  # 정상 이전 기록, 현재 슬롯은 아직 안 만듦
        for offset in (sh.SLOT_DELAY, 200, sh.SLOT_MAX_LATE):
            now = BASE + offset
            last = sh.load_last_slot(now, self.path)  # 가동 시작
            self.assertIsNone(sh.slot_due(now, last), offset)
        self.assertEqual(sh.slot_due(BASE + H + 100, sh.load_last_slot(BASE + 200, self.path)), BASE + H)

    def test_hostile_values_never_block_or_replay(self):
        now = BASE + 200
        for text in ('{"slot": 1e999}', '{"slot": 123456789012345678901234567890}', '{"slot": 3601}',
                     '{"slot": %d}' % (BASE + 5 * H), '{"slot": true}', 'null'):
            self.path.write_text(text, encoding='utf-8')
            self.assertEqual(sh.load_last_slot(now, self.path), BASE, text)

    def test_stale_signal_is_retried_without_claiming_then_made_once(self):
        calls = {'claim': [], 'make': 0, 'ready': False}
        def claim(slot): calls['claim'].append(slot)
        def make(): calls['make'] += 1
        last = BASE - H
        last, status = sh.plan_tick(BASE + 100, last, lambda: calls['ready'], claim, make)
        self.assertEqual((status, calls['claim'], last), ('waiting', [], BASE - H))
        calls['ready'] = True
        last, status = sh.plan_tick(BASE + 110, last, lambda: calls['ready'], claim, make)
        self.assertEqual((status, calls['claim'], calls['make'], last), ('made', [BASE], 1, BASE))
        last, status = sh.plan_tick(BASE + 120, last, lambda: True, claim, make)
        self.assertEqual((status, calls['make']), (None, 1))

    def test_crash_during_generation_loses_the_slot_but_never_duplicates_it(self):
        made = []
        def boom(): raise RuntimeError('생성 중 장애')
        import contextlib, io
        with contextlib.redirect_stderr(io.StringIO()):
            last, status = sh.plan_tick(BASE + 100, BASE - H, lambda: True, lambda s: sh.claim_slot(s, self.path), boom)
        self.assertEqual(status, 'failed')
        restarted = sh.load_last_slot(BASE + 160, self.path)
        last2, status2 = sh.plan_tick(BASE + 160, restarted, lambda: True, lambda s: None, lambda: made.append(1))
        self.assertEqual((status2, made), (None, []))

    def test_many_restarts_across_a_day_make_at_most_one_plan_per_slot(self):
        import random
        rng = random.Random(7)
        made_slots = []
        last = sh.load_last_slot(BASE, self.path)
        now = BASE
        for _ in range(24 * 120):  # 하루, 30초 간격 반복, 무작위 재시작·장애
            now += 30
            if rng.random() < 0.02:  # 재시작: 메모리 상태 소실
                last = sh.load_last_slot(now, self.path)
            def make(slot=sh.current_slot(now)):
                if rng.random() < 0.1:
                    raise RuntimeError('장애')
                made_slots.append(slot)
            import contextlib, io
            with contextlib.redirect_stderr(io.StringIO()):
                last, _ = sh.plan_tick(now, last, lambda: rng.random() > 0.2, lambda s: sh.claim_slot(s, self.path), make)
        self.assertEqual(len(made_slots), len(set(made_slots)))  # 슬롯당 최대 1회
        self.assertGreater(len(made_slots), 5)  # 장애·재시작이 있어도 일부 슬롯은 정상 생성(자명한 무동작 방지)


class RealProcesses(unittest.TestCase):
    """메모리를 공유하지 않는 실제 파이썬 프로세스로 영속 기록만을 통해 검증한다(강제 종료 포함)."""

    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.addCleanup(self.tmp.cleanup)
        self.path = Path(self.tmp.name) / 'plan_slot.json'
        self.log = Path(self.tmp.name) / 'made.log'

    def run_driver(self, start, now, mode):
        """start에 가동한 프로세스가 now에 한 번 반복한다(start 시점에 현재 슬롯은 처리된 것으로 본다)."""
        import subprocess, sys
        code = f'''
import os, sys
sys.path.insert(0, {str(Path(__file__).resolve().parent)!r})
from pathlib import Path
import shadow as sh
path, log, now = Path({str(self.path)!r}), Path({str(self.log)!r}), {now}
last = sh.load_last_slot({start}, path)
def claim(slot):
    sh.claim_slot(slot, path)
    if {mode!r} == 'crash_after_claim':
        os._exit(9)  # claim 직후 강제 종료(정상 종료 처리 없음)
def make():
    with log.open('a') as f: f.write(str(sh.current_slot(now)) + chr(10))
sh.plan_tick(now, last, lambda: True, claim, make)
'''
        return subprocess.run([sys.executable, '-c', code], capture_output=True, text=True, timeout=60).returncode

    def made(self):
        return self.log.read_text().split() if self.log.exists() else []

    def test_restart_in_late_window_then_exactly_one_plan_at_the_next_slot(self):
        sh.claim_slot(BASE - H, self.path)
        # 지연창 안(정각+200초)에 가동: 같은 시각에 반복해도 주문안 없음
        self.assertEqual(self.run_driver(BASE + 200, BASE + 200, 'normal'), 0)
        self.assertEqual(self.made(), [])
        # 그 프로세스가 계속 살아 다음 정규 슬롯에 반복하면 정확히 1회
        self.assertEqual(self.run_driver(BASE + 200, BASE + H + 100, 'normal'), 0)
        self.assertEqual(self.made(), [str(BASE + H)])
        # 같은 슬롯 안에서 다시 가동(재시작)하면 또 만들지 않는다
        self.assertEqual(self.run_driver(BASE + H + 130, BASE + H + 130, 'normal'), 0)
        self.assertEqual(self.made(), [str(BASE + H)])

    def test_force_kill_right_after_claim_never_recreates_the_slot(self):
        sh.claim_slot(BASE - H, self.path)
        # 정각 직후 가동한 프로세스가 다음 슬롯에서 claim 직후 강제 종료
        self.assertEqual(self.run_driver(BASE + 100, BASE + H + 100, 'crash_after_claim'), 9)
        self.assertEqual(self.made(), [])  # 기록만 되고 생성 전에 죽음
        # 재시작(같은 슬롯 안): 슬롯은 이미 기록됐고 가동 시작 시 현재 슬롯은 처리된 것으로 본다
        self.assertEqual(self.run_driver(BASE + H + 140, BASE + H + 140, 'normal'), 0)
        self.assertEqual(self.made(), [])
        # 그 프로세스가 다음 슬롯에 도달하면 정상 생성
        self.assertEqual(self.run_driver(BASE + H + 140, BASE + 2 * H + 100, 'normal'), 0)
        self.assertEqual(self.made(), [str(BASE + 2 * H)])

def note(stage,**detail):

                note('rollback_blocked_running_process',error=str(error));raise
            restore_owned_code(backup,old_owned)

                if target.exists() and target.parent.resolve()==APP.resolve():target.unlink()
            # Risk management only after rollback; do not silently revive old rotations.
            dp.patch_config(APP,{'live':{'auto_proposal':False}})
            dp.cancel_stop(APP);ok,why=dp.start_bot()
            note('rolled_back_regular_orders_paused',error=str(error),risk_management_restarted=ok)
            raise

"""안전 배포: 정본(ai-trade/v5) 수정 -> 테스트 -> 봇 정상 종료 -> 백업 -> 동기화 -> 재시작 -> 상태 확인 -> 실패 시 롤백.

  python tools/deploy_v5.py              점검만(봇을 건드리지 않음): 다른 파일, 테스트, 미체결 주문, 승인 상태
  python tools/deploy_v5.py --apply      사용자가 지금 배포를 승인한 것으로 보고 정지->적용->재시작
  python tools/deploy_v5.py --watch      정본이 바뀌면 자동 배포. 단, tools/APPROVED_COMMIT 의 커밋과 HEAD가 같을 때만

안전 규칙
- 프로세스를 강제로 죽이지 않는다. 봇의 정상 종료 파일(stop.txt 등)만 쓰고, 시간 안에 안 끝나면 정지 요청을 취소하고 중단한다.

- 재시작 후 상태 화면이 새로 만들어지지 않거나 오류가 보이면 백업으로 되돌리고 다시 시작한다.

def now():

def log(msg, app=None):
    line = f'[{now()}] {msg}'
    print(line)
    try:
        target = (app or APP) / 'data'
        target.mkdir(exist_ok=True)
        with (target / 'deploy.log').open('a', encoding='utf-8') as f:

def pid_alive(pid):
    # tasklist는 OEM 코드 페이지로 출력한다. 해석 못 하는 바이트가 있어도 stdout이 None이 되지 않게 replace로 읽는다.
    r = subprocess.run(['tasklist', '/FI', f'PID eq {pid}', '/FO', 'CSV', '/NH'], capture_output=True, text=True,
                       errors='replace', creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0))
    return f'"{pid}"' in (r.stdout or '')


def lock_alive(app, name):
    f = app / name
    if not f.exists():
        return False
    pid = f.read_text(errors='ignore').strip()
    return pid.isdigit() and pid_alive(pid)


def any_alive(app):
    return [n for n in LOCKS if lock_alive(app, n)]


def pending_orders(app):
    """미체결·미확정 주문이 있으면 이유 문자열, 없으면 ''. 장부가 없으면 ''."""
    f = app / 'data' / 'live' / 'live.json'
    if not f.exists():
        return ''
    d = json.loads(f.read_text(encoding='utf-8'))
    if d.get('pending_order'):
        return '미확정 주문 있음(pending_order)'
    if d.get('risk_pending_orders'):
        return '미확정 청산 주문 있음(risk_pending_orders)'
    return ''


def run_tests(src):
    """v5의 모든 test_*.py 모듈을 하나씩 실행한다. 반환: (통과 여부, 실패 목록)."""
    failed = []
    for f in sorted(src.glob('test_*.py')):
        try:
            r = subprocess.run([sys.executable, '-m', 'unittest', f.stem, '-q'], cwd=src, capture_output=True,
                               text=True, timeout=180, creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0))
            if r.returncode != 0:
                failed.append(f.stem)
        except subprocess.TimeoutExpired:
            failed.append(f.stem + '(시간 초과)')
    return not failed, failed


def backup(app):
    dst = app / '_backup' / dt.datetime.now().strftime('%Y%m%d_%H%M%S')
    dst.mkdir(parents=True)
    for f in list(app.glob('*.py')) + [app / 'README.md', app / 'config.json']:
        if f.exists():
            shutil.copyfile(f, dst / f.name)
    return dst


def restore(app, backup_dir):
    for f in backup_dir.iterdir():
        shutil.copyfile(f, app / f.name)


def patch_config(app, patch):
    """config.json에 중첩 dict를 덮어쓴다(예: {'live': {'trail_enabled': True}}). 호출자가 허용 키를 이미 검사한다."""
    f = app / 'config.json'
    cfg = json.loads(f.read_text(encoding='utf-8-sig'))
    for section, values in patch.items():
        cfg.setdefault(section, {}).update(values)
    f.write_text(json.dumps(cfg, ensure_ascii=False, indent=2), encoding='utf-8')


def request_stop(app):
    for n in STOPS:
        (app / n).write_text('deploy_v5 stop request\n', encoding='utf-8')


def cancel_stop(app):
    for n in STOPS:
        (app / n).unlink(missing_ok=True)


def wait_stopped(app, timeout):
    end = time.monotonic() + timeout
    while time.monotonic() < end:
        if not any_alive(app):
            return True
        time.sleep(2)
    return False


def start_bot(launch=None):
    """바탕화면 실행과 같은 경로(launch_v5.py)로 시작한다. 상태 화면이 새로 만들어질 때까지 기다린다."""
    launch = launch or LAUNCH
    # 자동 재시작이 매번 브라우저 탭을 새로 열지 않게 한다(화면은 이미 열려 있으면 자동 갱신된다).
    r = subprocess.run([sys.executable, str(launch), '--console'], cwd=launch.parent, capture_output=True, text=True,
                       timeout=420, creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0), env={**os.environ, 'V5_NO_BROWSER': '1'})
    return r.returncode == 0, (r.stderr or r.stdout)[-400:]


def start_errors(app, since):
    """시작 시각 이후 bot.stderr.log에 오류가 추가됐는지."""
    f = app / 'data' / 'bot.stderr.log'
    if not f.exists() or f.stat().st_mtime < since:
        return ''
    tail = f.read_text(encoding='utf-8', errors='replace')[-1500:]
    return tail if 'Traceback' in tail else ''


def preflight(app=None, src=None, with_tests=True):
    app, src = app or APP, src or SRC
    out = {'diff': sy.differences(), 'alive': any_alive(app), 'pending': pending_orders(app)}
    if with_tests:
        out['tests_ok'], out['tests_failed'] = run_tests(src)
    return out


def deploy(approved_by, app=None, src=None, launch=None, config_patch=None):
    app, src = app or APP, src or SRC
    pre = preflight(app, src)
    log(f'배포 요청({approved_by}): 다른 파일 {pre["diff"]} · 실행 중 {pre["alive"]} · 미확정 주문 {pre["pending"] or "없음"} · 테스트 {"통과" if pre["tests_ok"] else "실패 " + str(pre["tests_failed"])}', app)
    if not pre['diff'] and not config_patch:
        log('적용할 변경 없음', app)
        return 0
    if not pre['tests_ok']:
        log('테스트 실패: 배포하지 않음', app)
        return 1
    if pre['pending']:
        log('미확정 주문: 배포하지 않음. 주문이 정리된 뒤 다시 시도', app)
        return 1
    was_running = bool(pre['alive'])
    bk = backup(app)
    log(f'백업: {bk}', app)
    if was_running:
        request_stop(app)
        log('정상 종료 요청(강제 종료 없음). 최대 %d초 대기' % STOP_TIMEOUT, app)
        if not wait_stopped(app, STOP_TIMEOUT):
            cancel_stop(app)
            log('시간 안에 종료되지 않아 정지 요청을 취소하고 중단함. 봇은 그대로 둠', app)
            return 1
        if pending_orders(app):
            log('종료 후 미확정 주문 발견: 코드는 바꾸지 않고 그대로 재시작 시도', app)
            cancel_stop(app)
            start_bot(launch)
            return 1
    cancel_stop(app)
    started = time.time()
    for name in pre['diff']:
        shutil.copyfile(src / name, app / name)
    if config_patch:
        patch_config(app, config_patch)
        log(f'설정 변경 적용: {config_patch}', app)
    log(f'코드 {len(pre["diff"])}개 적용 (data·.env 그대로' + ('' if config_patch else ', config.json 그대로') + ')', app)
    ok, why = start_bot(launch) if was_running else (True, '')
    err = start_errors(app, started) if was_running else ''
    if was_running and (not ok or err):
        log(f'재시작 실패({why.strip()[:200]} {err[:200]}) -> 백업으로 롤백', app)
        restore(app, bk)
        ok2, why2 = start_bot(launch)
        log('롤백 후 재시작 ' + ('성공' if ok2 else '실패: ' + why2.strip()[:200] + ' — 수동 확인 필요'), app)
        return 2
    log('배포 완료' + ('' if was_running else ' (봇이 꺼져 있어 코드만 적용, 시작은 하지 않음)'), app)
    return 0


def head():
    r = subprocess.run(['git', 'rev-parse', 'HEAD'], cwd=ROOT, capture_output=True, text=True)
    return r.stdout.strip()


def v5_clean():
    r = subprocess.run(['git', 'status', '--porcelain', '--', 'v5'], cwd=ROOT, capture_output=True, text=True)
    return not r.stdout.strip()


def approved(head_hash):
    return APPROVAL.exists() and APPROVAL.read_text(encoding='utf-8').strip().split()[:1] == [head_hash]


def watch():
    log('감시 시작: 승인된 커밋만 자동 배포 (tools/APPROVED_COMMIT). 중지: Ctrl+C, 일시 정지: tools/deploy_pause.txt')
    while True:
        try:
            if not PAUSE.exists() and sy.differences() and v5_clean() and approved(head()):
                deploy('감시: 승인 커밋 ' + head()[:7])
        except Exception as e:  # 감시기는 죽지 않는다
            log(f'감시 오류: {type(e).__name__}: {e}')
        time.sleep(WATCH_SECONDS)


def main(argv=None):
    ap = argparse.ArgumentParser()
    ap.add_argument('--apply', action='store_true')
    ap.add_argument('--watch', action='store_true')
    ap.add_argument('--patch', help='config.json에 덮어쓸 설정 JSON 파일 경로(예: tools/patches/profit_protection.json). --apply와 함께 사용')
    a = ap.parse_args(argv)
    if not APP.exists():
        print('실행 사본 없음:', APP)
        return 1
    if a.watch:
        watch()
        return 0
    if a.apply:
        patch = json.loads(Path(a.patch).read_text(encoding='utf-8')) if a.patch else None
        return deploy('사용자 직접 실행' + (f' + 설정 {Path(a.patch).name}' if a.patch else ''), config_patch=patch)
    pre = preflight()
    print('다른 파일:', pre['diff'] or '없음')
    print('실행 중인 프로세스:', pre['alive'] or '없음')
    print('미확정 주문:', pre['pending'] or '없음')
    print('테스트:', '통과' if pre['tests_ok'] else '실패 ' + str(pre['tests_failed']))
    print('HEAD', head()[:7], '· 승인 상태:', '승인됨' if approved(head()) else '미승인')
    print('배포하려면: python tools/deploy_v5.py --apply')
    return 0


if __name__ == '__main__':
    sys.exit(main())

  python tools/sync_v5.py            무엇이 다른지만 출력
  python tools/sync_v5.py --apply    봇이 꺼져 있을 때만 복사 (봇 실행 중이면 거부)
규칙: 코드는 ai-trade에서만 고친다. 실행 사본은 이 도구로만 갱신한다.
"""
import argparse

from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent

DST = ROOT.parent / '코인 트레이딩' / 'v5-app' / 'v5'
SKIP = {'config.json'}


def digest(f):
    return hashlib.sha256(f.read_bytes().replace(b'\r\n', b'\n')).hexdigest()


def differences():
    out = []
    for f in sorted(SRC.glob('*.py')) + [SRC / 'README.md']:
        if f.name in SKIP or not f.exists():
            continue
        g = DST / f.name
        if not g.exists() or digest(f) != digest(g):
            out.append(f.name)
    return out


def bot_running():
    import re
    ps = ("$ErrorActionPreference='Stop'; [Console]::OutputEncoding=[Text.Encoding]::UTF8; "
          "Get-CimInstance Win32_Process | Where-Object { $_.Name -in 'python.exe','pythonw.exe' } "
          "| ForEach-Object { 'P:' + ($_.CommandLine -replace '[\\r\\n\\v\\f\\x1c-\\x1e\\x85\\u2028\\u2029]', ' ') }; 'END'")
    try:
        r = subprocess.run(['powershell', '-NoProfile', '-NonInteractive', '-Command', ps],
                           capture_output=True, text=True, encoding='utf-8', errors='replace', timeout=15)
    except (subprocess.TimeoutExpired, OSError, ValueError):
        raise RuntimeError('프로세스 조회 실패') from None
    if r.returncode != 0:
        raise RuntimeError('프로세스 조회 실패')
    lines = (r.stdout or '').lstrip('\ufeff').splitlines()
    if not lines or lines[-1].strip() != 'END':
        raise RuntimeError('프로세스 조회 출력 이상')
    pat = re.compile(r'(?i)(?:^|[\s"\'\\/])(?:bot|shadow)\.py(?=["\s]|$)')
    for line in lines[:-1]:
        if not line.startswith('P:'):
            raise RuntimeError('프로세스 조회 출력 이상')
        cmd = line[2:]
        if not cmd.strip():
            raise RuntimeError('프로세스 명령줄 조회 불명')
        if pat.search(cmd):
            return True
    return False


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--apply', action='store_true')
    a = ap.parse_args()
    if not DST.exists():
        print('실행 사본 없음:', DST)
        return 1
    diff = differences()
    print('다른 파일:', diff or '없음')
    if not a.apply or not diff:
        return 0
    if bot_running():
        print('봇이 실행 중입니다. 정지 후 다시 실행하세요 (python 코인 트레이딩/launch_v5.py --stop).')
        return 1
    for name in diff:
        shutil.copyfile(SRC / name, DST / name)
    print('복사 완료:', len(diff), '개 (config.json·data·.env는 그대로)')
    return 0


if __name__ == '__main__':
    sys.exit(main())
